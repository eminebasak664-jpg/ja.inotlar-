package com.jai.app.model;
public class Matrix {
    public final int rows, cols; public final float[] data;
    public Matrix(int r, int c) { this.rows=r; this.cols=c; this.data=new float[r*c]; }
    public float get(int r, int c) { return data[r*cols+c]; }
    public static Matrix mul(Matrix A, Matrix B) {
        Matrix C = new Matrix(A.rows, B.cols);
        for (int i=0; i<A.rows; i++) { for (int k=0; k<A.cols; k++) { float a=A.data[i*A.cols+k]; if(a==0) continue; for (int j=0; j<B.cols; j++) { C.data[i*C.cols+j] += a * B.data[k*B.cols+j]; } } }
        return C;
    }
}

package com.jai.app.model;
public class Int8Tensor {
    public final byte[] data; public final float scale; public final int zp;
    public Int8Tensor(byte[] d, float s, int z) { data=d; scale=s; zp=z; }
    public static Int8Tensor quantize(Matrix m) {
        float min=Float.MAX_VALUE, max=-Float.MAX_VALUE; for(float v:m.data){ if(v<min)min=v; if(v>max)max=v; }
        float s=(max-min)/255f; if(s==0)s=1e-8f; int z=Math.round(-min/s); if(z<0)z=0; if(z>255)z=255;
        byte[] q=new byte[m.data.length]; for(int i=0;i<m.data.length;i++) q[i]=(byte)(Math.round(m.data[i]/s)+z);
        return new Int8Tensor(q, s, z);
    }
}

public class BpeTokenizer {
    private Map<String, Integer> vocab = new HashMap<>();
    public BpeTokenizer(Context ctx) { try(BufferedReader r=new BufferedReader(new InputStreamReader(ctx.getAssets().open("vocab.txt")))) { String l; int i=0; while((l=r.readLine())!=null) vocab.put(l, i++); } catch(Exception e){} }
    public int[] encode(String t) { int[] ids=new int[t.length()]; for(int i=0;i<t.length();i++) ids[i]=vocab.getOrDefault(String.valueOf(t.charAt(i)), 0); return ids; }
    public String decode(int[] ids) { StringBuilder s=new StringBuilder(); for(int id:ids) { for(Map.Entry<String,Integer> e:vocab.entrySet()) { if(e.getValue()==id) { s.append(e.getKey()); break; } } } return s.toString(); }
}
