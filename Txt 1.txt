Dosya 6: .github/workflows/build.yml

```yaml
name: Build Ja.i APK

on:
  push:
    branches: [ main, master ]
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Set up JDK 17
        uses: actions/setup-java@v4
        with:
          java-version: '17'
          distribution: 'temurin'
      - name: Setup Android SDK
        uses: android-actions/setup-android@v3
      - name: Grant execute permission
        run: chmod +x gradlew || true
      - name: Build Debug APK
        run: ./gradlew assembleDebug --no-daemon
      - name: Upload APK
        uses: actions/upload-artifact@v4
        with:
          name: jai-debug-apk
          path: app/build/outputs/apk/debug/*.apk
      - name: Build Release APK
        if: github.ref == 'refs/heads/main'
        run: ./gradlew assembleRelease --no-daemon
      - name: Upload Release APK
        if: github.ref == 'refs/heads/main'
        uses: actions/upload-artifact@v4
        with:
          name: jai-release-apk
          path: app/build/outputs/apk/release/*.apk
```

Dosya 8: README.md

```markdown
# Ja.i — Kendi Zekâsıyla Çalışan Android Asistanı

## Hızlı Başlangıç

1. Bu repoyu GitHub'a yükle
2. GitHub Actions otomatik derler (`Actions` sekmesi)
3. `jai-debug-apk` artifact'ını indir
4. Telefona kur

## Model Eğitimi (Colab)

1. `colab/train_on_colab.ipynb` dosyasını Colab'a yükle
2. Çalıştır → `model_int8.bin` indir
3. `app/src/main/assets/model_int8.bin` olarak koy
4. Yeniden derle

## Bilgi Paketi

1. `colab/build_knowledge.ipynb` çalıştır → `knowledge.db` indir
2. `app/src/main/assets/knowledge_seed.db` olarak koy

## Mimari

- **Model**: 10 katman transformer, 576 hidden, 65M param, INT8
- **Öğrenme**: RAG (bilgi paketi ekleme)
- **Ses**: TTS offline (Android), STT online (Android)
- **Mesajlaşma**: WebSocket + kendi sunucu

## Lisans
Kişisel kullanım.
```

```java
package com.jai.app.model;

/**
 * Saf Java matris işlemleri. INT8 quantize destekli.
 * Yavaş ama bağımsız — JNI/NEON yok.
 */
public class Matrix {
    public final int rows, cols;
    public final float[] data; // row-major

    public Matrix(int rows, int cols) {
        this.rows = rows;
        this.cols = cols;
        this.data = new float[rows * cols];
    }

    public Matrix(int rows, int cols, float[] data) {
        this.rows = rows;
        this.cols = cols;
        this.data = data;
    }

    public float get(int r, int c) { return data[r * cols + c]; }
    public void set(int r, int c, float v) { data[r * cols + c] = v; }

    /** A(m×n) × B(n×p) = C(m×p) */
    public static Matrix mul(Matrix A, Matrix B) {
        if (A.cols != B.rows) throw new IllegalArgumentException(
            "Boyut uyumsuz: " + A.rows + "x" + A.cols + " × " + B.rows + "x" + B.cols);
        Matrix C = new Matrix(A.rows, B.cols);
        for (int i = 0; i < A.rows; i++) {
            int aRow = i * A.cols;
            int cRow = i * C.cols;
            for (int k = 0; k < A.cols; k++) {
                float a = A.data[aRow + k];
                if (a == 0) continue;
                int bRow = k * B.cols;
                for (int j = 0; j < B.cols; j++) {
                    C.data[cRow + j] += a * B.data[bRow + j];
                }
            }
        }
        return C;
    }

    public Matrix transpose() {
        Matrix T = new Matrix(cols, rows);
        for (int r = 0; r < rows; r++)
            for (int c = 0; c < cols; c++)
                T.data[c * rows + r] = data[r * cols + c];
        return T;
    }

    public Matrix add(Matrix B) {
        if (rows != B.rows || cols != B.cols) throw new IllegalArgumentException("add boyut");
        Matrix C = new Matrix(rows, cols);
        for (int i = 0; i < data.length; i++) C.data[i] = data[i] + B.data[i];
        return C;
    }

    public Matrix mulScalar(float s) {
        Matrix C = new Matrix(rows, cols);
        for (int i = 0; i < data.length; i++) C.data[i] = data[i] * s;
        return C;
    }

    public static Matrix softmax(Matrix X) {
        Matrix Y = new Matrix(X.rows, X.cols);
        for (int i = 0; i < X.rows; i++) {
            int row = i * X.cols;
            float max = Float.NEGATIVE_INFINITY;
            for (int j = 0; j < X.cols; j++) if (X.data[row + j] > max) max = X.data[row + j];
            float sum = 0;
            for (int j = 0; j < X.cols; j++) {
                float e = (float) Math.exp(X.data[row + j] - max);
                Y.data[row + j] = e;
                sum += e;
            }
            for (int j = 0; j < X.cols; j++) Y.data[row + j] /= sum;
        }
        return Y;
    }

    public static Matrix layerNorm(Matrix X, float[] gamma, float[] beta, float eps) {
        Matrix Y = new Matrix(X.rows, X.cols);
        for (int i = 0; i < X.rows; i++) {
            int row = i * X.cols;
            float mean = 0;
            for (int j = 0; j < X.cols; j++) mean += X.data[row + j];
            mean /= X.cols;
            float var = 0;
            for (int j = 0; j < X.cols; j++) {
                float d = X.data[row + j] - mean;
                var += d * d;
            }
            var /= X.cols;
            float inv = (float) (1.0 / Math.sqrt(var + eps));
            for (int j = 0; j < X.cols; j++) {
                float norm = (X.data[row + j] - mean) * inv;
                Y.data[row + j] = norm * gamma[j] + beta[j];
            }
        }
        return Y;
    }

    public static Matrix gelu(Matrix X) {
        Matrix Y = new Matrix(X.rows, X.cols);
        for (int i = 0; i < X.data.length; i++) {
            float x = X.data[i];
            Y.data[i] = 0.5f * x * (1f + (float) Math.tanh(
                0.7978845608f * (x + 0.044715f * x * x * x)));
        }
        return Y;
    }

    public static Matrix relu(Matrix X) {
        Matrix Y = new Matrix(X.rows, X.cols);
        for (int i = 0; i < X.data.length; i++) Y.data[i] = Math.max(0, X.data[i]);
        return Y;
    }
}
```

```java
package com.jai.app.model;

/**
 * INT8 quantize edilmiş ağırlık deposu.
 * Skala ve zero-point ile float32'ye geri açılır.
 */
public class Int8Tensor {
    public final int rows, cols;
    public final byte[] data;
    public final float scale;
    public final int zeroPoint;

    public Int8Tensor(int rows, int cols, byte[] data, float scale, int zeroPoint) {
        this.rows = rows;
        this.cols = cols;
        this.data = data;
        this.scale = scale;
        this.zeroPoint = zeroPoint;
    }

    public float getFloat(int r, int c) {
        int v = data[r * cols + c] & 0xFF;
        return (v - zeroPoint) * scale;
    }

    public Matrix dequantize() {
        Matrix m = new Matrix(rows, cols);
        for (int i = 0; i < data.length; i++) {
            int v = data[i] & 0xFF;
            m.data[i] = (v - zeroPoint) * scale;
        }
        return m;
    }

    /** Float matristen INT8'e quantize (min-max) */
    public static Int8Tensor quantize(Matrix m) {
        float min = Float.POSITIVE_INFINITY, max = Float.NEGATIVE_INFINITY;
        for (float v : m.data) {
            if (v < min) min = v;
            if (v > max) max = v;
        }
        float scale = (max - min) / 255f;
        if (scale == 0) scale = 1e-8f;
        int zp = Math.round(-min / scale);
        if (zp < 0) zp = 0; if (zp > 255) zp = 255;
        byte[] q = new byte[m.data.length];
        for (int i = 0; i < m.data.length; i++) {
            int v = Math.round(m.data[i] / scale) + zp;
            if (v < 0) v = 0; if (v > 255) v = 255;
            q[i] = (byte) v;
        }
        return new Int8Tensor(m.rows, m.cols, q, scale, zp);
    }
}
```

```java
package com.jai.app.model;

public class Dense {
    public final int inDim, outDim;
    public Matrix W;    // outDim × inDim
    public float[] b;   // outDim

    public Dense(int inDim, int outDim) {
        this.inDim = inDim;
        this.outDim = outDim;
        this.W = new Matrix(outDim, inDim);
        this.b = new float[outDim];
        // Xavier init
        float std = (float) Math.sqrt(2.0 / (inDim + outDim));
        java.util.Random rng = new java.util.Random(42);
        for (int i = 0; i < W.data.length; i++) W.data[i] = (float) (rng.nextGaussian() * std);
    }

    /** X: (n × inDim) → Y: (n × outDim) */
    public Matrix forward(Matrix X) {
        Matrix Y = Matrix.mul(X, W.transpose());
        for (int i = 0; i < Y.rows; i++)
            for (int j = 0; j < outDim; j++)
                Y.data[i * outDim + j] += b[j];
        return Y;
    }
}
```

```java
        float std = (float) Math.sqrt(1.0 / dim);
        for (int i = 0; i < W.data.length; i++) W.data[i] = (float) (rng.nextGaussian() * std);
    }

    /** tokenIds: (n) → (n × dim) */
    public Matrix forward(int[] tokenIds) {
        Matrix Y = new Matrix(tokenIds.length, dim);
        for (int i = 0; i < tokenIds.length; i++) {
            int id = tokenIds[i];
            if (id < 0 || id >= vocabSize) id = 0;
            System.arraycopy(W.data, id * dim, Y.data, i * dim, dim);
        }
        return Y;
    }
}
```

```java
package com.jai.app.model;

import android.content.Context;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.*;

/**
 * Basit BPE tokenizer. Vocab ve merge kuralları assets'ten yüklenir.
 * Vocab yoksa fallback: byte-level + whitespace.
 */
public class BpeTokenizer {
    private final Map<String, Integer> vocab = new HashMap<>();
    private final Map<Integer, String> reverse = new HashMap<>();
    private final List<int[]> merges = new ArrayList<>(); // [pairId1, pairId2, newId]
    private int nextId = 0;

    public BpeTokenizer(Context ctx) {
        try (BufferedReader r = new BufferedReader(new InputStreamReader(
                ctx.getAssets().open("vocab.txt")))) {
            String line;
            while ((line = r.readLine()) != null) {
                String[] p = line.split("\t");
                if (p.length == 2) {
                    vocab.put(p[0], Integer.parseInt(p[1]));
                    reverse.put(Integer.parseInt(p[1]), p[0]);
                }
            }
        } catch (Exception e) {
            // fallback
            for (int i = 0; i < 256; i++) {
                vocab.put("byte_" + i, nextId);
                reverse.put(nextId++, "byte_" + i);
            }
        }
        for (int i = 0; i < 1000; i++) { // placeholder merges
            merges.add(new int[]{i, i + 1, 256 + i});
        }
    }

    public int[] encode(String text) {
        if (text == null || text.isEmpty()) return new int[]{0};
        byte[] bytes = text.getBytes(java.nio.charset.StandardCharsets.UTF_8);
        int[] ids = new int[bytes.length];
        for (int i = 0; i < bytes.length; i++) {
            Integer id = vocab.get("byte_" + (bytes[i] & 0xFF));
            ids[i] = id != null ? id : 0;
        }
        return ids;
    }

    public String decode(int[] ids) {
        StringBuilder sb = new StringBuilder();
        for (int id : ids) {
            String tok = reverse.get(id);
            if (tok != null && tok.startsWith("byte_")) {
                try {
                    int b = Integer.parseInt(tok.substring(5));
                    sb.append((char) b);
                } catch (Exception ignored) {}
            } else if (tok != null) {
                sb.append(tok);
            }
        }
        return sb.toString();
    }

    public int vocabSize() { return Math.max(vocab.size(), 16000); }
}
```

```java
package com.jai.app.model;

import java.util.Random;

public class Sampler {
    private final Random rng;
    public float temperature = 0.8f;
    public int topK = 40;
    public float topP = 0.9f;

    public Sampler() { this.rng = new Random(); }
    public Sampler(long seed) { this.rng = new Random(seed); }

    public int sample(float[] logits) {
        // temperature
        float[] scaled = new float[logits.length];
        for (int i = 0; i < logits.length; i++) scaled[i] = logits[i] / temperature;

        // top-k
        Integer[] idx = new Integer[scaled.length];
        for (int i = 0; i < idx.length; i++) idx[i] = i;
        Arrays.sort(idx, (a, b) -> Float.compare(scaled[b], scaled[a]));
        int k = Math.min(topK, scaled.length);
        float[] probs = new float[k];
        float max = scaled[idx[0]];
        float sum = 0;
        for (int i = 0; i < k; i++) {
            probs[i] = (float) Math.exp(scaled[idx[i]] - max);
            sum += probs[i];
        }
        for (int i = 0; i < k; i++) probs[i] /= sum;

        // top-p
        float cum = 0;
        int cutoff = k;
        for (int i = 0; i < k; i++) {
            cum += probs[i];
            if (cum >= topP) { cutoff = i + 1; break; }
        }
        cum = 0;
        for (int i = 0; i < cutoff; i++) cum += probs[i];

        float r = rng.nextFloat() * cum;
        float acc = 0;
        for (int i = 0; i < cutoff; i++) {
            acc += probs[i];
            if (r <= acc) return idx[i];
        }
        return idx[0];
    }
    private static class Arrays {
        static <T> void sort(T[] a, java.util.Comparator<T> c) {
            java.util.Arrays.sort(a, c);
        }
    }
}
```

```java
package com.jai.app.model;

public class MultiHeadAttention {
    public final int dim, numHeads, headDim;
    public final Dense qProj, kProj, vProj, oProj;

    public MultiHeadAttention(int dim, int numHeads) {
        this.dim = dim;
        this.numHeads = numHeads;
        this.headDim = dim / numHeads;
        this.qProj = new Dense(dim, dim);
        this.kProj = new Dense(dim, dim);
        this.vProj = new Dense(dim, dim);
        this.oProj = new Dense(dim, dim);
    }

    public Matrix forward(Matrix X) {
        int seq = X.rows;
        Matrix Q = qProj.forward(X);
        Matrix K = kProj.forward(X);
        Matrix V = vProj.forward(X);

        float scale = (float) (1.0 / Math.sqrt(headDim));
        Matrix out = new Matrix(seq, dim);

        for (int h = 0; h < numHeads; h++) {
            int off = h * headDim;
            // scores: (seq × seq)
            Matrix scores = new Matrix(seq, seq);
            for (int i = 0; i < seq; i++) {
                for (int j = 0; j < seq; j++) {
                    float s = 0;
                    for (int d = 0; d < headDim; d++)
                        s += Q.data[i * dim + off + d] * K.data[j * dim + off + d];
                    scores.data[i * seq + j] = s * scale;
                }
            }
            Matrix attn = Matrix.softmax(scores);
            // out_head = attn × V_head
            for (int i = 0; i < seq; i++) {
                for (int d = 0; d < headDim; d++) {
                    float s = 0;
                    for (int j = 0; j < seq; j++)
                        s += attn.data[i * seq + j] * V.data[j * dim + off + d];
                    out.data[i * dim + off + d] = s;
                }
            }
        }
        return oProj.forward(out);
    }
}
```

```java
public class TransformerBlock {
    public final LayerNorm ln1, ln2;
    public final MultiHeadAttention attn;
    public final FeedForward ffn;

    public TransformerBlock(int dim, int heads, int ffnHidden) {
        this.ln1 = new LayerNorm(dim);

        this.ffn = new FeedForward(dim, ffnHidden);

        Matrix attnOut = attn.forward(a);
        Matrix x1 = X.add(attnOut);

        Matrix ffnOut = ffn.forward(b);
```

```java
package com.jai.app.model;

/**
 * Tam model: embedding + N transformer bloğu + çıkış projection.
 */
public class TransformerModel {
    public final int vocabSize, dim, numLayers, numHeads, ffnHidden, maxSeq;
    public final Embedding tokenEmb;
    public final Embedding posEmb;
    public final TransformerBlock[] blocks;
    public final LayerNorm finalNorm;
    public final Dense lmHead;

    public TransformerModel(int vocabSize, int dim, int numLayers, int numHeads,
                             int ffnHidden, int maxSeq) {
        this.vocabSize = vocabSize;
        this.dim = dim;
        this.numLayers = numLayers;
        this.numHeads = numHeads;
        this.ffnHidden = ffnHidden;
        this.maxSeq = maxSeq;
        this.tokenEmb = new Embedding(vocabSize, dim);
        this.posEmb = new Embedding(maxSeq, dim);
        this.blocks = new TransformerBlock[numLayers];
        for (int i = 0; i < numLayers; i++) blocks[i] = new TransformerBlock(dim, numHeads, ffnHidden);
        this.finalNorm = new LayerNorm(dim);
        this.lmHead = new Dense(dim, vocabSize);
    }

    public Matrix forward(int[] tokenIds) {
        int seq = Math.min(tokenIds.length, maxSeq);
        Matrix X = tokenEmb.forward(tokenIds);
        int[] positions = new int[seq];
        for (int i = 0; i < seq; i++) positions[i] = i;
        Matrix P = posEmb.forward(positions);
        X = X.add(P);
        for (TransformerBlock b : blocks) X = b.forward(X);
        X = finalNorm.forward(X);
        return lmHead.forward(X); // (seq × vocabSize) = logits
    }
}
```

```java
    public String generate(String prompt, int maxTokens) {
        int[] ids = tokenizer.encode(prompt);
        StringBuilder out = new StringBuilder();
        Sampler sampler = new Sampler();
        int[] ctxIds = new int[Math.min(ids.length, model.maxSeq)];
        for (int i = 0; i < ctxIds.length; i++) ctxIds[i] = ids[ids.length - ctxIds.length + i];

            Matrix logits = model.forward(ctxIds); // (seq × vocab)
            int last = logits.rows - 1;
            float[] lastLogits = new float[logits.cols];
            for (int i = 0; i < logits.cols; i++) lastLogits[i] = logits.data[last * logits.cols + i];
            int nextTok = sampler.sample(lastLogits);
            if (nextTok == 2) break; // EOS placeholder
            out.append(tokenizer.decode(new int[]{nextTok}));

            int[] newCtx = new int[Math.min(ctxIds.length + 1, model.maxSeq)];
            int skip = Math.max(0, ctxIds.length + 1 - model.maxSeq);
            for (int i = 0; i < ctxIds.length - skip; i++) newCtx[i] = ctxIds[i + skip];
            newCtx[newCtx.length - 1] = nextTok;
            ctxIds = newCtx;
        }
        return out.toString();
    }
```

```java
package com.jai.app.trainer;

import com.jai.app.model.Matrix;

/**
 * Adam optimizer. Parametreleri yerinde günceller.
 * Sunucuda/Colab'da çalışacak trainer için.
 */
public class Optimizer {
    public final float lr, beta1, beta2, eps;
    private final java.util.Map<Matrix, float[]> m = new java.util.IdentityHashMap<>();
    private final java.util.Map<Matrix, float[]> v = new java.util.IdentityHashMap<>();
    private int step = 0;

    public Optimizer(float lr) {
        this.lr = lr; this.beta1 = 0.9f; this.beta2 = 0.999f; this.eps = 1e-8f;
    }

    public void step(Matrix param, Matrix grad) {
        float[] mm = m.computeIfAbsent(param, k -> new float[param.data.length]);
        float[] vv = v.computeIfAbsent(param, k -> new float[param.data.length]);
        step++;
        float bc1 = 1 - (float) Math.pow(beta1, step);
        float bc2 = 1 - (float) Math.pow(beta2, step);
        for (int i = 0; i < param.data.length; i++) {
            mm[i] = beta1 * mm[i] + (1 - beta1) * grad.data[i];
            vv[i] = beta2 * vv[i] + (1 - beta2) * grad.data[i] * grad.data[i];
            float mHat = mm[i] / bc1;
            float vHat = vv[i] / bc2;
            param.data[i] -= lr * mHat / ((float) Math.sqrt(vHat) + eps);
        }
    }
}
```

```java
package com.jai.app.trainer;

import com.jai.app.model.Matrix;

public class Loss {
    /** Cross-entropy: logits (n × vocab), hedefler (n) */
    public static float crossEntropy(Matrix logits, int[] targets) {
        float total = 0;
        for (int i = 0; i < logits.rows; i++) {
            int row = i * logits.cols;
            float max = Float.NEGATIVE_INFINITY;
            for (int j = 0; j < logits.cols; j++) if (logits.data[row + j] > max) max = logits.data[row + j];
            float sum = 0;
            for (int j = 0; j < logits.cols; j++) sum += (float) Math.exp(logits.data[row + j] - max);
            float logProb = logits.data[row + targets[i]] - max - (float) Math.log(sum);
            total -= logProb;
        }
        return total / logits.rows;
    }

    /** Cross-entropy gradyanı: dL/dlogits */
    public static Matrix crossEntropyGrad(Matrix logits, int[] targets) {
        Matrix grad = new Matrix(logits.rows, logits.cols);
        for (int i = 0; i < logits.rows; i++) {
            int row = i * logits.cols;
            float max = Float.NEGATIVE_INFINITY;
            for (int j = 0; j < logits.cols; j++) if (logits.data[row + j] > max) max = logits.data[row + j];
            float sum = 0;
            for (int j = 0; j < logits.cols; j++) sum += (float) Math.exp(logits.data[row + j] - max);
            for (int j = 0; j < logits.cols; j++) {
                float p = (float) Math.exp(logits.data[row + j] - max) / sum;
                grad.data[row + j] = p;
            }
            grad.data[row + targets[i]] -= 1;
            for (int j = 0; j < logits.cols; j++) grad.data[row + j] /= logits.rows;
        }
        return grad;
    }
}
```

```java
import com.jai.app.model.Matrix;
import com.jai.app.model.TransformerModel;

/**
 * Eğitim döngüsü. FP32 ağırlıklar. Sunucuda/Colab'da çalışır.
 * Telefonda koşmaz; ağır.
 */
public class Trainer {
    public final TransformerModel model;
    public final Optimizer opt;
    public float lastLoss = 0;

    public Trainer(TransformerModel model, float lr) {
        this.model = model;
        this.opt = new Optimizer(lr);
    }

    public void trainStep(int[] inputIds, int[] targetIds) {
        // forward
        Matrix logits = model.forward(inputIds); // (seq × vocab)
        int seq = Math.min(inputIds.length, logits.rows);
        Matrix subLogits = new Matrix(seq, logits.cols);
        for (int i = 0; i < seq; i++)
            System.arraycopy(logits.data, i * logits.cols, subLogits.data, i * logits.cols, logits.cols);
        int[] targets = new int[seq];
        System.arraycopy(targetIds, 0, targets, 0, seq);

        float loss = Loss.crossEntropy(subLogits, targets);
        lastLoss = loss;

        // backward (basitleştirilmiş: sadece lmHead güncellenir)
        Matrix grad = Loss.crossEntropyGrad(subLogits, targets);
        // lmHead.W: (vocab × dim), grad: (seq × vocab)
        // dW = grad^T × sonHidden
        // Tam backprop burada değil, sadece çıkış katmanı (demo)
        // Gerçek eğitim Colab notebook'unda tam türev ile yapılır.

        // lmHead bias güncelle
        float[] bGrad = new float[model.vocabSize];
        for (int i = 0; i < seq; i++)
            for (int j = 0; j < model.vocabSize; j++)
                bGrad[j] += grad.data[i * model.vocabSize + j];
        Matrix bg = new Matrix(1, model.vocabSize);
        for (int j = 0; j < model.vocabSize; j++) bg.data[j] = bGrad[j];
        // (Bias için ayrı optimizer gerekir; basitlik için burada duruyoruz)
    }
}
```

```java
            }
            writeFloats(out, m.finalNorm.gamma); writeFloats(out, m.finalNorm.beta);
            writeMatrix(out, m.lmHead.W); writeFloats(out, m.lmHead.b);
        }
    }

    private static void writeMatrix(DataOutputStream o, Matrix m) throws IOException {
        o.writeInt(m.rows); o.writeInt(m.cols);
        for (float v : m.data) o.writeFloat(v);
    }
    private static void writeFloats(DataOutputStream o, float[] a) throws IOException {
        o.writeInt(a.length);
        for (float v : a) o.writeFloat(v);
    }
}
```
