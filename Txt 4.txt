    private static volatile AppDatabase INSTANCE;

    public abstract UserDao users();
    public abstract KnowledgeDao knowledge();
    public abstract LearningTaskDao learningTasks();

            synchronized (AppDatabase.class) {
                if (INSTANCE == null) {
                    INSTANCE = Room.databaseBuilder(ctx.getApplicationContext(), AppDatabase.class, "jai.db").fallbackToDestructiveMigration().build();
                }
            }
        }
        return INSTANCE;
    }

package com.jai.app.model;
public class Matrix {
    public final int rows, cols;
    public final float[] data;
    public Matrix(int r, int c) { this.rows = r; this.cols = c; this.data = new float[r * c]; }
    public Matrix(int r, int c, float[] d) { this.rows = r; this.cols = c; this.data = d; }
    public float get(int r, int c) { return data[r * cols + c]; }
    public void set(int r, int c, float v) { data[r * cols + c] = v; }
    public static Matrix mul(Matrix A, Matrix B) {
        if (A.cols != B.rows) throw new IllegalArgumentException("Boyut uyumsuz");
        Matrix C = new Matrix(A.rows, B.cols);
        for (int i = 0; i < A.rows; i++) {
            for (int k = 0; k < A.cols; k++) {
                float a = A.data[i * A.cols + k];
                if (a == 0) continue;
                for (int j = 0; j < B.cols; j++) C.data[i * C.cols + j] += a * B.data[k * B.cols + j];
            }
        }
        return C;
    }
    public Matrix transpose() {
        Matrix T = new Matrix(cols, rows);
        for (int r = 0; r < rows; r++) for (int c = 0; c < cols; c++) T.data[c * rows + r] = data[r * cols + c];
        return T;
    }
    public Matrix add(Matrix B) {
        Matrix C = new Matrix(rows, cols);
        for (int i = 0; i < data.length; i++) C.data[i] = data[i] + B.data[i];
        return C;
    }
    public static Matrix softmax(Matrix X) {
        Matrix Y = new Matrix(X.rows, X.cols);
        for (int i = 0; i < X.rows; i++) {
            int row = i * X.cols; float max = Float.NEGATIVE_INFINITY;
            for (int j = 0; j < X.cols; j++) if (X.data[row + j] > max) max = X.data[row + j];
            float sum = 0;
            for (int j = 0; j < X.cols; j++) { float e = (float) Math.exp(X.data[row + j] - max); Y.data[row + j] = e; sum += e; }
            for (int j = 0; j < X.cols; j++) Y.data[row + j] /= sum;
        }
        return Y;
    }
    public static Matrix layerNorm(Matrix X, float[] gamma, float[] beta, float eps) {
        Matrix Y = new Matrix(X.rows, X.cols);
        for (int i = 0; i < X.rows; i++) {
            int row = i * X.cols; float mean = 0;
            for (int j = 0; j < X.cols; j++) mean += X.data[row + j];
            mean /= X.cols;
            float var = 0;
            for (int j = 0; j < X.cols; j++) { float d = X.data[row + j] - mean; var += d * d; }
            var /= X.cols; float inv = (float) (1.0 / Math.sqrt(var + eps));
            for (int j = 0; j < X.cols; j++) Y.data[row + j] = (X.data[row + j] - mean) * inv * gamma[j] + beta[j];
        }
        return Y;
    }
    public static Matrix gelu(Matrix X) {
        Matrix Y = new Matrix(X.rows, X.cols);
        for (int i = 0; i < X.data.length; i++) { float x = X.data[i]; Y.data[i] = 0.5f * x * (1f + (float) Math.tanh(0.7978845608f * (x + 0.044715f * x * x * x))); }
        return Y;
    }
}

package com.jai.app.model;
public class Int8Tensor {
    public final int rows, cols;
    public final byte[] data;
    public final float scale;
    public final int zeroPoint;
    public Int8Tensor(int r, int c, byte[] d, float s, int zp) { rows = r; cols = c; data = d; scale = s; zeroPoint = zp; }
    public Matrix dequantize() {
        Matrix m = new Matrix(rows, cols);
        for (int i = 0; i < data.length; i++) m.data[i] = ((data[i] & 0xFF) - zeroPoint) * scale;
        return m;
    }
    public static Int8Tensor quantize(Matrix m) {
        float min = Float.MAX_VALUE, max = -Float.MAX_VALUE;
        for (float v : m.data) { if (v < min) min = v; if (v > max) max = v; }
        float scale = (max - min) / 255f; if (scale == 0) scale = 1e-8f;
        int zp = Math.round(-min / scale); if (zp < 0) zp = 0; if (zp > 255) zp = 255;
        byte[] q = new byte[m.data.length];
        for (int i = 0; i < m.data.length; i++) { int v = Math.round(m.data[i] / scale) + zp; if (v < 0) v = 0; if (v > 255) v = 255; q[i] = (byte) v; }
        return new Int8Tensor(m.rows, m.cols, q, scale, zp);
    }
}

package com.jai.app.model;
public class Dense {
    public final int inDim, outDim;
    public Matrix W;
    public float[] b;
    public Dense(int inDim, int outDim) { this.inDim = inDim; this.outDim = outDim; this.W = new Matrix(outDim, inDim); this.b = new float[outDim]; }
    public Matrix forward(Matrix X) {
        Matrix Y = Matrix.mul(X, W.transpose());
        for (int i = 0; i < Y.rows; i++) for (int j = 0; j < outDim; j++) Y.data[i * outDim + j] += b[j];
        return Y;
    }
}

package com.jai.app.model;
public class Embedding {
    public final int vocabSize, dim;
    public Matrix W;
    public Embedding(int vocabSize, int dim) { this.vocabSize = vocabSize; this.dim = dim; this.W = new Matrix(vocabSize, dim); }
    public Matrix forward(int[] tokenIds) {
        Matrix Y = new Matrix(tokenIds.length, dim);
        for (int i = 0; i < tokenIds.length; i++) {
            int id = tokenIds[i]; if (id < 0 || id >= vocabSize) id = 0;
            System.arraycopy(W.data, id * dim, Y.data, i * dim, dim);
        }
        return Y;
    }
}

public class MultiHeadAttention {
    public final int dim, numHeads, headDim;
    public final Dense qProj, kProj, vProj, oProj;
    public MultiHeadAttention(int dim, int numHeads) {
        this.dim = dim; this.numHeads = numHeads; this.headDim = dim / numHeads;
        this.qProj = new Dense(dim, dim); this.kProj = new Dense(dim, dim); this.vProj = new Dense(dim, dim); this.oProj = new Dense(dim, dim);
    }
    public Matrix forward(Matrix X) {
        int seq = X.rows;
        Matrix Q = qProj.forward(X); Matrix K = kProj.forward(X); Matrix V = vProj.forward(X);
        float scale = (float) (1.0 / Math.sqrt(headDim));
        Matrix out = new Matrix(seq, dim);
        for (int h = 0; h < numHeads; h++) {
            int off = h * headDim;
            Matrix scores = new Matrix(seq, seq);
            for (int i = 0; i < seq; i++) {
                for (int j = 0; j < seq; j++) {
                    float s = 0;
                    for (int d = 0; d < headDim; d++) s += Q.data[i * dim + off + d] * K.data[j * dim + off + d];
                    scores.data[i * seq + j] = s * scale;
                }
            }
            Matrix attn = Matrix.softmax(scores);
            for (int i = 0; i < seq; i++) {
                for (int d = 0; d < headDim; d++) {
                    float s = 0;
                    for (int j = 0; j < seq; j++) s += attn.data[i * seq + j] * V.data[j * dim + off + d];
                    out.data[i * dim + off + d] = s;
                }
            }
        }
        return oProj.forward(out);
    }
}

    public final FeedForward ffn;
    public TransformerBlock(int dim, int heads, int ffnHidden) {
        this.ln1 = new LayerNorm(dim); this.attn = new MultiHeadAttention(dim, heads);
        this.ln2 = new LayerNorm(dim); this.ffn = new FeedForward(dim, ffnHidden);
    }
    public Matrix forward(Matrix X) {

        Matrix b = ln2.forward(x1); Matrix ffnOut = ffn.forward(b); return x1.add(ffnOut);

package com.jai.app.model;
public class TransformerModel {
    public final int vocabSize, dim, numLayers, numHeads, ffnHidden, maxSeq;
    public final Embedding tokenEmb, posEmb;
    public final TransformerBlock[] blocks;
    public final LayerNorm finalNorm;
    public final Dense lmHead;
    public TransformerModel(int vocabSize, int dim, int numLayers, int numHeads, int ffnHidden, int maxSeq) {
        this.vocabSize = vocabSize; this.dim = dim; this.numLayers = numLayers; this.numHeads = numHeads; this.ffnHidden = ffnHidden; this.maxSeq = maxSeq;
        this.tokenEmb = new Embedding(vocabSize, dim); this.posEmb = new Embedding(maxSeq, dim);
        this.blocks = new TransformerBlock[numLayers];
        for (int i = 0; i < numLayers; i++) blocks[i] = new TransformerBlock(dim, numHeads, ffnHidden);
        this.finalNorm = new LayerNorm(dim); this.lmHead = new Dense(dim, vocabSize);
    }
    public Matrix forward(int[] tokenIds) {
        int seq = Math.min(tokenIds.length, maxSeq);
        Matrix X = tokenEmb.forward(tokenIds);
        int[] positions = new int[seq]; for (int i = 0; i < seq; i++) positions[i] = i;
        X = X.add(posEmb.forward(positions));
        for (TransformerBlock b : blocks) X = b.forward(X);
        X = finalNorm.forward(X);
        return lmHead.forward(X);
    }
}

package com.jai.app.model;
import android.content.Context;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.HashMap;
import java.util.Map;
public class BpeTokenizer {
    private final Map<String, Integer> vocab = new HashMap<>();
    private final Map<Integer, String> reverse = new HashMap<>();
    public BpeTokenizer(Context ctx) {
        try (BufferedReader r = new BufferedReader(new InputStreamReader(ctx.getAssets().open("vocab.txt")))) {
            String line; int i = 0;
            while ((line = r.readLine()) != null) { vocab.put(line, i); reverse.put(i, line); i++; }
        } catch (Exception e) {
            for (int i = 0; i < 256; i++) { vocab.put("b" + i, i); reverse.put(i, "b" + i); }
        }
    }
    public int[] encode(String text) {
        byte[] bytes = text.getBytes(java.nio.charset.StandardCharsets.UTF_8);
        int[] ids = new int[bytes.length];
        for (int i = 0; i < bytes.length; i++) ids[i] = vocab.getOrDefault("b" + (bytes[i] & 0xFF), 0);
        return ids;
    }
    public String decode(int[] ids) {
        StringBuilder sb = new StringBuilder();
        for (int id : ids) {
            String tok = reverse.get(id);
            if (tok != null && tok.startsWith("b")) { try { sb.append((char) Integer.parseInt(tok.substring(1))); } catch (Exception ignored) {} }
        }
        return sb.toString();
    }
}

// DOSYA 49: app/src/main/java/com/jai/app/model/InferenceRuntime.java
package com.jai.app.model;
import android.content.Context;
import java.io.InputStream;
public class InferenceRuntime {
    public TransformerModel model;
    public BpeTokenizer tokenizer;
    private String state = "NOT_TRAINED";
    public void init(Context ctx) {
        tokenizer = new BpeTokenizer(ctx);
        model = new TransformerModel(16000, 576, 10, 8, 2304, 1024);
        try (InputStream in = ctx.getAssets().open("model_int8.bin")) {
            state = "READY";
        } catch (Exception e) { state = "NOT_TRAINED"; }
    }
    public String getState() { return state; }
    public boolean isReady() { return state.equals("READY"); }
    public String generate(String prompt, int maxTokens) {
        if (!isReady()) return "Model yüklenmedi: " + state;
        int[] ids = tokenizer.encode(prompt);
        StringBuilder out = new StringBuilder();
        Sampler sampler = new Sampler();
        int[] ctxIds = new int[Math.min(ids.length, model.maxSeq)];
        for (int i = 0; i < ctxIds.length; i++) ctxIds[i] = ids[ids.length - ctxIds.length + i];
        for (int step = 0; step < maxTokens; step++) {
            Matrix logits = model.forward(ctxIds);
            int last = logits.rows - 1;
            float[] lastLogits = new float[logits.cols];
            for (int i = 0; i < logits.cols; i++) lastLogits[i] = logits.data[last * logits.cols + i];
            int nextTok = sampler.sample(lastLogits);
            if (nextTok == 2) break;
            out.append(tokenizer.decode(new int[]{nextTok}));
            int[] newCtx = new int[Math.min(ctxIds.length + 1, model.maxSeq)];
            int skip = Math.max(0, ctxIds.length + 1 - model.maxSeq);
            for (int i = 0; i < ctxIds.length - skip; i++) newCtx[i] = ctxIds[i + skip];
            newCtx[newCtx.length - 1] = nextTok;
            ctxIds = newCtx;
        }
        return out.toString();
    }
}
