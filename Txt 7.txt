// DOSYA 56: app/src/main/java/com/jai/app/brain/BrainEngine.java
package com.jai.app.brain;
import com.jai.app.model.InferenceRuntime;
import com.jai.app.memory.RagEngine;
import com.jai.app.tools.ToolRegistry;
import com.jai.app.tools.ToolResult;
import java.util.HashMap;
import java.util.Locale;
import java.util.Map;

public class BrainEngine {
    public static final String MODE_FLASH = "FLASH", MODE_THINK = "THINK";
    public static final String EFFORT_FAST = "FAST", EFFORT_MID = "MID", EFFORT_MAX = "MAX", EFFORT_DEEP = "DEEP";
    
    public enum Intent {
        MATH, TIME, MEMORY_SEARCH, FILE_READ, FILE_WRITE, ZIP, WEB_REQUEST, LEARNING_REQUEST, GREETING, THANKS, WHO_ARE_YOU, HELP, UNKNOWN
    }
    
    public static class Result {
        public String reply, status, trace;
        public Result(String r, String s, String t) { reply = r; status = s; trace = t; }
    }

    private final InferenceRuntime runtime;
    private final RagEngine rag;

    public BrainEngine(InferenceRuntime runtime, RagEngine rag) {
        this.runtime = runtime; this.rag = rag;
    }

    public Intent classify(String input) {
        if (input == null) return Intent.UNKNOWN;
        String t = input.trim().toLowerCase(new Locale("tr", "TR"));
        if (t.isEmpty()) return Intent.UNKNOWN;
        if (t.matches("^(merhaba|selam|hey|hi|hello|günaydın).*")) return Intent.GREETING;
        if (t.matches("^(teşekkür|sağ ?ol|thanks).*")) return Intent.THANKS;
        if (t.contains("sen kimsin") || t.contains("adın ne")) return Intent.WHO_ARE_YOU;
        if (t.contains("yardım") || t.contains("ne yapabilirsin")) return Intent.HELP;
        if (t.contains("saat kaç") || t.contains("tarih ne")) return Intent.TIME;
        if (t.contains("hatırlıyor musun") || t.contains("hafızanda")) return Intent.MEMORY_SEARCH;
        if (t.contains("dosya") && (t.contains("oluştur") || t.contains("yaz") || t.contains("kaydet"))) return Intent.FILE_WRITE;
        if (t.contains("zip") || t.contains("arşiv")) return Intent.ZIP;
        if (t.contains("araştır") || t.contains("internette") || t.contains("web'de")) return Intent.WEB_REQUEST;
        if (t.contains("öğren")) return Intent.LEARNING_REQUEST;
        if (t.matches("^[0-9+\\-*/().,\\s]+$") && t.matches(".*[+\\-*/].*")) return Intent.MATH;
        if (t.contains("hesapla") || t.contains("kaç eder")) return Intent.MATH;
        return Intent.UNKNOWN;
    }

    public Result respond(String userInput, String mode, String effort, boolean searchOn, ToolRegistry tools) {
        Intent intent = classify(userInput);
        StringBuilder trace = new StringBuilder().append("Amaç: ").append(intent.name()).append('\n');

        switch (intent) {
            case MATH: {
                Map<String, Object> in = new HashMap<>(); in.put("expr", userInput);
                ToolResult r = run(tools, "math", in, trace);
                return new Result(r.ok ? "Sonuç: " + r.output : "Çözemedim.", r.ok ? "OK" : "PARTIAL", trace.toString());
            }
            case TIME: {
                ToolResult r = run(tools, "clock", new HashMap<>(), trace);
                return new Result(r.ok ? "Şu an: " + r.output : "Hata", r.ok ? "OK" : "ERROR", trace.toString());
            }
            case MEMORY_SEARCH: {
                Map<String, Object> in = new HashMap<>(); in.put("query", userInput);
                ToolResult r = run(tools, "memory_search", in, trace);
                return new Result("Hafıza:\n" + (r.ok ? r.output : "hata"), r.ok ? "OK" : "ERROR", trace.toString());
            }
            case WEB_REQUEST:
                if (!searchOn) return new Result("Web gerekli ama Ara kapalı.", "PARTIAL", trace.toString());
                return new Result("__NEEDS_WEB__" + userInput, "OK", trace.toString());
            case LEARNING_REQUEST:
                return new Result("__NEEDS_LEARNING__" + userInput, "OK", trace.toString());
            case GREETING: return new Result("Merhaba.", "OK", trace.toString());
            case THANKS: return new Result("Rica ederim.", "OK", trace.toString());
            case WHO_ARE_YOU: return new Result("Ben Ja.i. Model: " + runtime.getState(), "OK", trace.toString());
            case HELP: return new Result("Aritmetik, saat, hafıza, web (Ara açıkken), öğrenme kuyruğu çalışır.", "OK", trace.toString());
            default: {
                if (runtime.isReady()) {
                    String ragCtx = rag.buildContext(userInput);
                    String prompt = (ragCtx.isEmpty() ? "" : "[context]\n" + ragCtx + "[/context]\n") + "Kullanıcı: " + userInput + "\nJa.i:";
                    String out = runtime.generate(prompt, 128);
                    return new Result(out.isEmpty() ? "(boş)" : out, "OK", trace.toString());
                }
                Map<String, Object> in = new HashMap<>(); in.put("query", userInput);
                ToolResult r = run(tools, "memory_search", in, trace);
                if (r.ok && r.output != null && !r.output.contains("bulunamadı")) return new Result("Hafızamda var:\n" + r.output, "OK", trace.toString());
                return new Result("Bilmiyorum. Model eğitilmedi. (Ara açıksa web'den bakabilirim.)", "PARTIAL", trace.toString());
            }
        }
    }

    private ToolResult run(ToolRegistry tools, String id, Map<String, Object> input, StringBuilder trace) {
        try {
            var t = tools.get(id);
            if (t == null) return ToolResult.err("araç yok");
            ToolResult r = t.execute(input);
            trace.append("→ ").append(id).append(r.ok ? " ✓\n" : " ✗\n");
            return r;
        } catch (Exception e) {
            trace.append("✗ ").append(e.getMessage()).append('\n');
            return ToolResult.err(e.getMessage());
        }
    }
}

package com.jai.app.brain;
public final class ExpressionEvaluator {
    public static Double eval(String raw) {
        if (raw == null) return null;
        String s = raw.replace(",", ".").replace(" ", "");
        if (s.isEmpty()) return null;
        try {
            Parser p = new Parser(s);
            double v = p.parseExpression();
            if (p.pos != s.length()) return null;
            if (Double.isNaN(v) || Double.isInfinite(v)) return null;
            return v;
        } catch (RuntimeException e) { return null; }
    }
    private static final class Parser {
        final String s; int pos = 0;
        Parser(String s) { this.s = s; }
        double parseExpression() {
            double v = parseTerm();
            while (pos < s.length() && (s.charAt(pos) == '+' || s.charAt(pos) == '-')) {
                char op = s.charAt(pos++); double r = parseTerm();
                v = (op == '+') ? v + r : v - r;
            }
            return v;
        }
        double parseTerm() {
            double v = parseFactor();
            while (pos < s.length() && (s.charAt(pos) == '*' || s.charAt(pos) == '/')) {
                char op = s.charAt(pos++); double r = parseFactor();
                if (op == '*') v *= r; else { if (r == 0) throw new ArithmeticException(); v /= r; }
            }
            return v;
        }
        double parseFactor() {
            if (pos < s.length() && s.charAt(pos) == '(') {
                pos++; double v = parseExpression();
                if (pos >= s.length() || s.charAt(pos) != ')') throw new RuntimeException();
                pos++; return v;
            }
            if (pos < s.length() && s.charAt(pos) == '-') { pos++; return -parseFactor(); }
            int start = pos;
            while (pos < s.length() && (Character.isDigit(s.charAt(pos)) || s.charAt(pos) == '.')) pos++;
            if (start == pos) throw new RuntimeException();
            return Double.parseDouble(s.substring(start, pos));
        }
    }
}

package com.jai.app.tools;
import android.content.Context;
import android.net.Uri;
import androidx.core.content.FileProvider;
import java.io.File;
import java.io.FileOutputStream;
import java.io.InputStream;
import java.util.List;
import java.util.Map;
import java.util.zip.ZipEntry;
import java.util.zip.ZipOutputStream;
public class ZipTool implements Tool {
    private final Context ctx;
    public ZipTool(Context ctx) { this.ctx = ctx; }
    public String id() { return "zip"; }
    public String name() { return "ZIP"; }
    @SuppressWarnings("unchecked")
    public ToolResult execute(Map<String, Object> input) throws Exception {
        List<String> uris = (List<String>) input.get("uris");
        if (uris == null || uris.isEmpty()) return ToolResult.err("uri yok");
        File dir = new File(ctx.getFilesDir(), "outputs"); dir.mkdirs();
        File zf = new File(dir, "archive_" + System.currentTimeMillis() + ".zip");
        try (ZipOutputStream zos = new ZipOutputStream(new FileOutputStream(zf))) {
            for (String u : uris) {
                String name = Uri.parse(u).getLastPathSegment();
                if (name == null) name = "file";
                name = name.replace("..", "_").replace("/", "_").replace("\\", "_");
                zos.putNextEntry(new ZipEntry(name));
                try (InputStream in = ctx.getContentResolver().openInputStream(Uri.parse(u))) {
                    if (in != null) { byte[] buf = new byte[8192]; int r; while ((r = in.read(buf)) > 0) zos.write(buf, 0, r); }
                }
                zos.closeEntry();
            }
        }
        return ToolResult.ok(FileProvider.getUriForFile(ctx, ctx.getPackageName() + ".files", zf).toString());
    }
}

    public String register(String name, String surname, String email, int age) {

        prefs.edit().putLong("uid", id).apply();
        return "OK";

        long id = prefs.getLong("uid", -1);
        if (id < 0) return null;
        for (User u : db.users().all()) if (u.id == id) return u;
        return null;
    }

    public void setAdmin(boolean admin) { prefs.edit().putBoolean("is_admin", admin).apply(); }

package com.jai.app.video;
import android.graphics.Bitmap;
import android.media.MediaCodec;
import android.media.MediaCodecInfo;
import android.media.MediaFormat;
import android.media.MediaMuxer;
import java.io.File;
import java.nio.ByteBuffer;

public class VideoEncoder {
    private MediaCodec codec; private MediaMuxer muxer; private int track; private boolean muxerStarted;
    
    public VideoEncoder(File dir, int w, int h) throws Exception {
        MediaFormat fmt = MediaFormat.createVideoFormat("video/avc", w, h);
        fmt.setInteger(MediaFormat.KEY_COLOR_FORMAT, MediaCodecInfo.CodecCapabilities.COLOR_FormatYUV420Flexible);
        fmt.setInteger(MediaFormat.KEY_BIT_RATE, 4000000); fmt.setInteger(MediaFormat.KEY_FRAME_RATE, 24); fmt.setInteger(MediaFormat.KEY_I_FRAME_INTERVAL, 1);
        codec = MediaCodec.createEncoderByType("video/avc"); codec.configure(fmt, null, null, MediaCodec.CONFIGURE_FLAG_ENCODE); codec.start();
        muxer = new MediaMuxer(new File(dir, "video.mp4").getAbsolutePath(), MediaMuxer.OutputFormat.MUXER_OUTPUT_MPEG_4);
    }

    public void writeFrame(Bitmap bmp, int idx) throws Exception {
        MediaCodec.BufferInfo info = new MediaCodec.BufferInfo();
        int inIdx = codec.dequeueInputBuffer(10000);
        if (inIdx >= 0) {
            ByteBuffer buf = codec.getInputBuffer(inIdx);
            if (buf != null) {
                int[] pixels = new int[bmp.getWidth() * bmp.getHeight()]; bmp.getPixels(pixels, 0, bmp.getWidth(), 0, 0, bmp.getWidth(), bmp.getHeight());
                byte[] yuv = rgbToYuv(pixels, bmp.getWidth(), bmp.getHeight()); buf.clear(); buf.put(yuv);
                codec.queueInputBuffer(inIdx, 0, yuv.length, idx * 33333L, 0);
            }
        }
        int outIdx = codec.dequeueOutputBuffer(info, 10000);
        if (outIdx >= 0) {
            ByteBuffer outBuf = codec.getOutputBuffer(outIdx);
            if (outBuf != null && info.size > 0 && !muxerStarted) {
                MediaFormat outFmt = codec.getOutputFormat(); track = muxer.addTrack(outFmt); muxer.start(); muxerStarted = true;
            }
            if (muxerStarted && outBuf != null) { outBuf.position(info.offset); outBuf.limit(info.offset + info.size); muxer.writeSampleData(track, outBuf, info); }
            codec.releaseOutputBuffer(outIdx, false);
        }
    }

    public void finish() {
        if (codec != null) { codec.stop(); codec.release(); }
        if (muxer != null && muxerStarted) { muxer.stop(); muxer.release(); }
    }

    private byte[] rgbToYuv(int[] rgb, int w, int h) {
        byte[] yuv = new byte[w * h * 3 / 2];
        for (int i = 0; i < w * h; i++) { int r = (rgb[i] >> 16) & 0xff, g = (rgb[i] >> 8) & 0xff, b = rgb[i] & 0xff; yuv[i] = (byte) (0.299 * r + 0.587 * g + 0.114 * b); }
        return yuv;
    }
}

import com.jai.app.data.KnowledgeEntity;

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import java.net.URLEncoder;
public class WebResearchEngine {
    private final AppDatabase db;
    public WebResearchEngine(AppDatabase db) { this.db = db; }
    public ToolResult search(String query) {
        try {
            Document doc = Jsoup.connect("https://html.duckduckgo.com/html/?q=" + URLEncoder.encode(query, "UTF-8")).userAgent("Mozilla/5.0").timeout(10000).get();
            StringBuilder sb = new StringBuilder(); int count = 0;
            for (Element a : doc.select("a.result__a")) {
                String title = a.text(); String href = a.attr("href");
                String snip = "";
                if (a.parent() != null && a.parent().selectFirst(".result__snippet") != null) snip = a.parent().selectFirst(".result__snippet").text();
                sb.append("• ").append(title).append("\n  ").append(snip).append("\n\n");
                db.knowledge().insert(new KnowledgeEntity(query, title + " - " + snip, href));
                if (++count >= 5) break;
            }
            return count == 0 ? ToolResult.ok("(Sonuç yok)") : ToolResult.ok(sb.toString());
        } catch (Exception e) { return ToolResult.err(e.getMessage()); }
    }
}
