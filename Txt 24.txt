// DOSYA 270: colab/train_image_model.ipynb
{
 "cells": [
  {
   "cell_type": "code",
   "source": [
    "!pip install torch torchvision\n",
    "import torch, torch.nn as nn, torch.optim as optim\n",
    "from torchvision import datasets, transforms\n",
    "from torch.utils.data import DataLoader\n",
    "print('CUDA:', torch.cuda.is_available())"
   ]
  },
  {
   "cell_type": "code",
   "source": [
    "transform = transforms.Compose([transforms.Resize(64), transforms.ToTensor()])\n",
    "dataset = datasets.CIFAR10(root='./data', download=True, transform=transform)\n",
    "dataloader = DataLoader(dataset, batch_size=32, shuffle=True)\n",
    "class SmallUNet(nn.Module):\n",
    "    def __init__(self):\n",
    "        super().__init__()\n",
    "        self.enc1 = nn.Conv2d(3, 16, 3, padding=1)\n",
    "        self.enc2 = nn.Conv2d(16, 32, 3, padding=1)\n",
    "        self.dec1 = nn.Conv2d(32, 16, 3, padding=1)\n",
    "        self.dec2 = nn.Conv2d(16, 3, 3, padding=1)\n",
    "    def forward(self, x):\n",
    "        x = torch.relu(self.enc1(x))\n",
    "        x = torch.relu(self.enc2(x))\n",
    "        x = torch.relu(self.dec1(x))\n",
    "        return torch.sigmoid(self.dec2(x))\n",
    "model = SmallUNet().cuda()\n",
    "opt = optim.Adam(model.parameters(), lr=0.001)\n",
    "loss_fn = nn.MSELoss()\n",
    "for epoch in range(3):\n",
    "    for step, (imgs, _) in enumerate(dataloader):\n",
    "        imgs = imgs.cuda()\n",
    "        out = model(imgs)\n",
    "        loss = loss_fn(out, imgs)\n",
    "        opt.zero_grad(); loss.backward(); opt.step()\n",
    "        if step % 100 == 0: print(f'Epoch {epoch}, Step {step}, Loss {loss.item():.4f}')"
   ]
  },
  {
   "cell_type": "code",
   "source": [
    "import struct\n",
    "def q(w):\n",
    "    mn, mx = w.min().item(), w.max().item()\n",
    "    s = (mx - mn) / 255 if mx > mn else 1e-8\n",
    "    z = max(0, min(255, int(round(-mn / s))))\n",
    "    b = ((w / s) + z).round().clamp(0, 255).byte().numpy().tobytes()\n",
    "    return b, s, z\n",
    "with open('image_model_int8.bin', 'wb') as f:\n",
    "    sd = model.state_dict()\n",
    "    for name in ['enc1.weight', 'enc2.weight', 'dec1.weight', 'dec2.weight']:\n",
    "        mat = sd[name].cpu()\n",
    "        b, s, z = q(mat)\n",
    "        f.write(struct.pack('>ii', mat.shape[0], mat.shape[1]))\n",
    "        f.write(struct.pack('>f', s))\n",
    "        f.write(struct.pack('>i', z))\n",
    "        f.write(b)\n",
