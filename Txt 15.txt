// DOSYA 241: app/src/main/java/com/jai/app/video/AudioTrackWriter.java
package com.jai.app.video;

import android.media.MediaCodec;
import android.media.MediaFormat;
import android.media.MediaMuxer;
import java.nio.ByteBuffer;

public class AudioTrackWriter {
    private MediaCodec codec;
    private MediaMuxer muxer;
    private int trackIndex = -1;

    public void init(MediaMuxer muxer) throws Exception {
        this.muxer = muxer;
        MediaFormat format = MediaFormat.createAudioFormat("audio/mp4a-latm", 44100, 2);
        format.setInteger(MediaFormat.KEY_BIT_RATE, 64000);
        format.setInteger(MediaFormat.KEY_AAC_PROFILE, MediaCodecInfo.CodecProfileLevel.AACObjectLC);
        codec = MediaCodec.createEncoderByType("audio/mp4a-latm");
        codec.configure(format, null, null, MediaCodec.CONFIGURE_FLAG_ENCODE);
        codec.start();
    }

    public void writeSilence(long presentationTimeUs) {
        try {
            int inIdx = codec.dequeueInputBuffer(1000);
            if (inIdx >= 0) {
                ByteBuffer buffer = codec.getInputBuffer(inIdx);
                if (buffer != null) {
                    buffer.clear();
                    byte[] silence = new byte[1024]; // Basit sessizlik tamponu
                    buffer.put(silence);
                    codec.queueInputBuffer(inIdx, 0, silence.length, presentationTimeUs, 0);
                }
            }
            MediaCodec.BufferInfo info = new MediaCodec.BufferInfo();
            int outIdx = codec.dequeueOutputBuffer(info, 1000);
            if (outIdx >= 0) {
                if (trackIndex < 0) {
                    trackIndex = muxer.addTrack(codec.getOutputFormat());
                }
                ByteBuffer outBuf = codec.getOutputBuffer(outIdx);
                if (outBuf != null) {
                    muxer.writeSampleData(trackIndex, outBuf, info);
                }
                codec.releaseOutputBuffer(outIdx, false);
            }
        } catch (Exception ignored) {}
    }

    public void finish() {
        try {
            codec.stop();
            codec.release();
        } catch (Exception ignored) {}
    }
}

// DOSYA 242: app/src/main/java/com/jai/app/video/VideoEncoder.java
package com.jai.app.video;
import android.graphics.Bitmap;
import android.media.MediaCodec;
import android.media.MediaCodecInfo;
import android.media.MediaFormat;
import android.media.MediaMuxer;
import java.io.File;
import java.nio.ByteBuffer;

public class VideoEncoder {
    private MediaCodec codec; private MediaMuxer muxer; private int track; private boolean muxerStarted;
    private AudioTrackWriter audioWriter;
    
    public VideoEncoder(File dir, int w, int h) throws Exception {
        MediaFormat fmt = MediaFormat.createVideoFormat("video/avc", w, h);
        fmt.setInteger(MediaFormat.KEY_COLOR_FORMAT, MediaCodecInfo.CodecCapabilities.COLOR_FormatYUV420Flexible);
        fmt.setInteger(MediaFormat.KEY_BIT_RATE, 4000000); fmt.setInteger(MediaFormat.KEY_FRAME_RATE, 24); fmt.setInteger(MediaFormat.KEY_I_FRAME_INTERVAL, 1);
        codec = MediaCodec.createEncoderByType("video/avc"); codec.configure(fmt, null, null, MediaCodec.CONFIGURE_FLAG_ENCODE); codec.start();
        muxer = new MediaMuxer(new File(dir, "video.mp4").getAbsolutePath(), MediaMuxer.OutputFormat.MUXER_OUTPUT_MPEG_4);
        audioWriter = new AudioTrackWriter();
        audioWriter.init(muxer);
    }

    public void writeFrame(Bitmap bmp, int idx) throws Exception {
        MediaCodec.BufferInfo info = new MediaCodec.BufferInfo();
        int inIdx = codec.dequeueInputBuffer(10000);
        if (inIdx >= 0) {
            ByteBuffer buf = codec.getInputBuffer(inIdx);
            if (buf != null) {
                int[] pixels = new int[bmp.getWidth() * bmp.getHeight()]; bmp.getPixels(pixels, 0, bmp.getWidth(), 0, 0, bmp.getWidth(), bmp.getHeight());
                byte[] yuv = rgbToYuv(pixels, bmp.getWidth(), bmp.getHeight()); buf.clear(); buf.put(yuv);
                codec.queueInputBuffer(inIdx, 0, yuv.length, idx * 33333L, 0);
                // Her kare için sessiz ses yaz
                audioWriter.writeSilence(idx * 33333L);
            }
        }
        int outIdx = codec.dequeueOutputBuffer(info, 10000);
        if (outIdx >= 0) {
            ByteBuffer outBuf = codec.getOutputBuffer(outIdx);
            if (outBuf != null && info.size > 0 && !muxerStarted) {
                MediaFormat outFmt = codec.getOutputFormat(); track = muxer.addTrack(outFmt); muxer.start(); muxerStarted = true;
            }
            if (muxerStarted && outBuf != null) { outBuf.position(info.offset); outBuf.limit(info.offset + info.size); muxer.writeSampleData(track, outBuf, info); }
            codec.releaseOutputBuffer(outIdx, false);
        }
    }

    public void finish() {
        if (codec != null) { codec.stop(); codec.release(); }
        if (audioWriter != null) audioWriter.finish();
        if (muxer != null && muxerStarted) { muxer.stop(); muxer.release(); }
    }

    private byte[] rgbToYuv(int[] rgb, int w, int h) {
        byte[] yuv = new byte[w * h * 3 / 2];
        for (int i = 0; i < w * h; i++) { int r = (rgb[i] >> 16) & 0xff, g = (rgb[i] >> 8) & 0xff, b = rgb[i] & 0xff; yuv[i] = (byte) (0.299 * r + 0.587 * g + 0.114 * b); }
        return yuv;
    }
}

package com.jai.app.ui;
import android.net.Uri;
import android.os.Bundle;
import android.widget.MediaController;
import android.widget.VideoView;
import androidx.appcompat.app.AppCompatActivity;
import com.jai.app.R;
import java.io.File;
public class VideoPlayerActivity extends AppCompatActivity {
    @Override protected void onCreate(Bundle b) {
        super.onCreate(b); 
        setContentView(R.layout.activity_video_player);
        VideoView vv = findViewById(R.id.videoView);
        String path = getIntent().getStringExtra("path");
        if (path != null && new File(path).exists()) {
            MediaController mc = new MediaController(this);
            mc.setAnchorView(vv);
            vv.setMediaController(mc);
            vv.setVideoURI(Uri.fromFile(new File(path)));
            vv.start();
        }
    }
}

// DOSYA 255: app/src/main/java/com/jai/app/image/GenerativeImageEngine.java
package com.jai.app.image;
import android.content.Context;
import android.graphics.Bitmap;
import android.graphics.Color;
import com.jai.app.model.image.ImageModel;
import com.jai.app.model.Matrix;
import com.jai.app.trainer.PtqImageReader;
import java.io.InputStream;

public class GenerativeImageEngine {
    private ImageModel model;
    private Context ctx;
    public GenerativeImageEngine(Context ctx) {
        this.ctx = ctx;
        this.model = null;
        try (InputStream in = ctx.getAssets().open("image_model_int8.bin")) {
            this.model = PtqImageReader.read(in);
        } catch (Exception ignored) {}
    }
    public Bitmap generate(String prompt) {
        if (model == null) return new WebImageEngine(ctx).generate(prompt);
        int size = 64;
        Matrix input = new Matrix(size * size, 3);
        java.util.Random r = new java.util.Random(prompt.hashCode());
        for (int i = 0; i < input.data.length; i++) input.data[i] = r.nextFloat();
        Matrix out = model.forward(input, size, size);
        Bitmap bmp = Bitmap.createBitmap(size, size, Bitmap.Config.ARGB_8888);
        int[] pixels = new int[size * size];
        for (int i = 0; i < pixels.length && i * 3 + 2 < out.data.length; i++) {
            int rCol = Math.max(0, Math.min(255, (int)(out.data[i*3] * 255)));
            int gCol = Math.max(0, Math.min(255, (int)(out.data[i*3+1] * 255)));
            int bCol = Math.max(0, Math.min(255, (int)(out.data[i*3+2] * 255)));
            pixels[i] = Color.rgb(rCol, gCol, bCol);
        }
        bmp.setPixels(pixels, 0, size, 0, 0, size, size);
        return Bitmap.createScaledBitmap(bmp, 512, 512, false);
    }
}
