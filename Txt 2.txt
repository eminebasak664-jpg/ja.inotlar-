```java
package com.jai.app.trainer;

import com.jai.app.model.Matrix;
import com.jai.app.model.TransformerModel;

/**
 * Post-Training Quantization kalibrasyonu.
 * Modeli FP32'den INT8'e çevirirken ağırlık dağılımını ölçer.
 * Kalibrasyon seti: birkaç yüz örnek cümle.
 */
public class PtqCalibrator {
    public static class Range {
        public float min = Float.POSITIVE_INFINITY;
        public float max = Float.NEGATIVE_INFINITY;
        public void observe(float v) {
            if (v < min) min = v;
            if (v > max) max = v;
        }
        public float scale() {
            float s = (max - min) / 255f;
            return s == 0 ? 1e-8f : s;
        }
        public int zeroPoint() {
            int zp = Math.round(-min / scale());
            return Math.max(0, Math.min(255, zp));
        }
    }

    public static void observe(TransformerModel m, Range[] ranges) {
        int i = 0;
        for (float v : m.tokenEmb.W.data) { ranges[i++].observe(v); }
        for (float v : m.posEmb.W.data) { ranges[i++].observe(v); }
        for (var b : m.blocks) {
            for (float v : b.attn.qProj.W.data) ranges[i++].observe(v);
            for (float v : b.attn.kProj.W.data) ranges[i++].observe(v);
            for (float v : b.attn.vProj.W.data) ranges[i++].observe(v);
            for (float v : b.attn.oProj.W.data) ranges[i++].observe(v);
            for (float v : b.ffn.fc1.W.data) ranges[i++].observe(v);
            for (float v : b.ffn.fc2.W.data) ranges[i++].observe(v);
        }
        for (float v : m.lmHead.W.data) ranges[i++].observe(v);
    }

    public static int countMatrices(TransformerModel m) {
        return 2 + m.numLayers * 6 + 1;
    }
}
```

```java
package com.jai.app.trainer;

import com.jai.app.model.Int8Tensor;
import com.jai.app.model.Matrix;
import com.jai.app.model.TransformerModel;

import java.io.*;

/**
 * FP32 modeli INT8 binary formata yazar.
 * Format: header + her ağırlık matrisi için (rows, cols, scale, zp, byte[])
 */
public class PtqWriter {
    public static void write(TransformerModel m, String path) throws IOException {
        try (DataOutputStream out = new DataOutputStream(
                new BufferedOutputStream(new FileOutputStream(path)))) {
            out.writeInt(0x4A414931); // "JAI1"
            out.writeInt(m.vocabSize);
            out.writeInt(m.dim);
            out.writeInt(m.numLayers);
            out.writeInt(m.numHeads);
            out.writeInt(m.ffnHidden);
            out.writeInt(m.maxSeq);

            writeInt8(out, m.tokenEmb.W);
            writeInt8(out, m.posEmb.W);
            for (var b : m.blocks) {
                writeInt8(out, b.attn.qProj.W);
                writeInt8(out, b.attn.kProj.W);
                writeInt8(out, b.attn.vProj.W);
                writeInt8(out, b.attn.oProj.W);
                writeInt8(out, b.ffn.fc1.W);
                writeInt8(out, b.ffn.fc2.W);
                // LayerNorm gamma/beta FP16 gibi davran (float olarak)
                writeFloats(out, b.ln1.gamma);
                writeFloats(out, b.ln1.beta);
                writeFloats(out, b.ln2.gamma);
                writeFloats(out, b.ln2.beta);
                writeFloats(out, b.ffn.fc1.b);
                writeFloats(out, b.ffn.fc2.b);
                writeFloats(out, b.attn.qProj.b);
                writeFloats(out, b.attn.kProj.b);
                writeFloats(out, b.attn.vProj.b);
                writeFloats(out, b.attn.oProj.b);
            }
            writeInt8(out, m.lmHead.W);
            writeFloats(out, m.finalNorm.gamma);
            writeFloats(out, m.finalNorm.beta);
            writeFloats(out, m.lmHead.b);
        }
    }

    private static void writeInt8(DataOutputStream out, Matrix m) throws IOException {
        Int8Tensor t = Int8Tensor.quantize(m);
        out.writeInt(t.rows);
        out.writeInt(t.cols);
        out.writeFloat(t.scale);
        out.writeInt(t.zeroPoint);
        out.write(t.data);
    }

    private static void writeFloats(DataOutputStream out, float[] a) throws IOException {
        out.writeInt(a.length);
        for (float v : a) out.writeFloat(v);
    }
}
```

```java
package com.jai.app.trainer;

import com.jai.app.model.TransformerModel;

import java.io.*;

/**
 * INT8 binary model dosyasını okur, TransformerModel'e yükler.
 * InferenceRuntime tarafından çağrılır.
 */
public class PtqReader {
    public static TransformerModel read(InputStream in) throws IOException {
        DataInputStream din = new DataInputStream(new BufferedInputStream(in));
        int magic = din.readInt();
        if (magic != 0x4A414931) throw new IOException("Geçersiz model dosyası");
        int vocab = din.readInt();
        int dim = din.readInt();
        int layers = din.readInt();
        int heads = din.readInt();
        int ffn = din.readInt();
        int maxSeq = din.readInt();

        TransformerModel m = new TransformerModel(vocab, dim, layers, heads, ffn, maxSeq);

        readInt8(din, m.tokenEmb.W);
        readInt8(din, m.posEmb.W);
        for (var b : m.blocks) {
            readInt8(din, b.attn.qProj.W);
            readInt8(din, b.attn.kProj.W);
            readInt8(din, b.attn.vProj.W);
            readInt8(din, b.attn.oProj.W);
            readInt8(din, b.ffn.fc1.W);
            readInt8(din, b.ffn.fc2.W);
            b.ln1.gamma = readFloats(din);
            b.ln1.beta = readFloats(din);
            b.ln2.gamma = readFloats(din);
            b.ln2.beta = readFloats(din);
            b.ffn.fc1.b = readFloats(din);
            b.ffn.fc2.b = readFloats(din);
            b.attn.qProj.b = readFloats(din);
            b.attn.kProj.b = readFloats(din);
            b.attn.vProj.b = readFloats(din);
            b.attn.oProj.b = readFloats(din);
        }
        readInt8(din, m.lmHead.W);
        m.finalNorm.gamma = readFloats(din);
        m.finalNorm.beta = readFloats(din);
        m.lmHead.b = readFloats(din);

        return m;
    }

    private static void readInt8(DataInputStream in, com.jai.app.model.Matrix target) throws IOException {
        int rows = in.readInt();
        int cols = in.readInt();
        float scale = in.readFloat();
        int zp = in.readInt();
        byte[] data = new byte[rows * cols];
        in.readFully(data);
        for (int i = 0; i < data.length; i++) {
            int v = data[i] & 0xFF;
            target.data[i] = (v - zp) * scale;
        }
    }

    private static float[] readFloats(DataInputStream in) throws IOException {
        int n = in.readInt();
        float[] a = new float[n];
        for (int i = 0; i < n; i++) a[i] = in.readFloat();
        return a;
    }
}
```

```java
package com.jai.app.sr;

/**
 * 512→1024 süper çözünürlük. Küçük CNN (3 katman konvolüsyon).
 * Ağırlıklar rastgele init; gerçek eğitim Colab'da.
 * Telefonda INT8 weightlerle çalışır.
 */
public class SRModel {
    public final int inSize = 512;
    public final int outSize = 1024;
    // Basit 3 katman: [1→16, 16→32, 32→3]
    public final float[][] conv1 = new float[16][9];
    public final float[] bias1 = new float[16];
    public final float[][] conv2 = new float[32][9 * 16];
    public final float[] bias2 = new float[32];
    public final float[][] conv3 = new float[3][9 * 32];
    public final float[] bias3 = new float[3];

    public SRModel() {
        java.util.Random rng = new java.util.Random(7);
        for (var a : conv1) for (int i = 0; i < a.length; i++) a[i] = (float) (rng.nextGaussian() * 0.1);
        for (var a : conv2) for (int i = 0; i < a.length; i++) a[i] = (float) (rng.nextGaussian() * 0.05);
        for (var a : conv3) for (int i = 0; i < a.length; i++) a[i] = (float) (rng.nextGaussian() * 0.05);
    }

    /** Basit nearest-neighbor büyütme — hızlı, ağırlık kullanmaz.
     *  Gerçek SR için ağırlıklar kullanılır ama M21'de yavaş olur. */
    public int[] upscale(int[] input, int w, int h) {
        int outW = w * 2;
        int outH = h * 2;
        int[] output = new int[outW * outH];
        for (int y = 0; y < outH; y++) {
            for (int x = 0; x < outW; x++) {
                int sx = x / 2;
                int sy = y / 2;
                output[y * outW + x] = input[sy * w + sx];
            }
        }
        return output;
    }
}
```

```java
package com.jai.app.sr;

import android.graphics.Bitmap;
import android.graphics.BitmapFactory;
import java.io.File;
import java.io.FileOutputStream;

public class SRPipeline {
    private final SRModel model;
    public SRPipeline(SRModel m) { this.model = m; }

    public Bitmap upscale(Bitmap src) {
        int w = src.getWidth();
        int h = src.getHeight();
        int[] pixels = new int[w * h];
        src.getPixels(pixels, 0, w, 0, 0, w, h);
        int[] up = model.upscale(pixels, w, h);
        Bitmap out = Bitmap.createBitmap(w * 2, h * 2, Bitmap.Config.ARGB_8888);
        out.setPixels(up, 0, w * 2, 0, 0, w * 2, h * 2);
        return out;
    }

    public boolean upscaleFile(String srcPath, String dstPath) {
        try {
            Bitmap src = BitmapFactory.decodeFile(srcPath);
            if (src == null) return false;
            Bitmap out = upscale(src);
            try (FileOutputStream fos = new FileOutputStream(new File(dstPath))) {
                out.compress(Bitmap.CompressFormat.PNG, 95, fos);
            }
            src.recycle();
            out.recycle();
            return true;
        } catch (Exception e) {
            return false;
        }
    }
}
```

```java
package com.jai.app.image;

import android.graphics.Bitmap;
import android.graphics.Canvas;
import android.graphics.Color;
import android.graphics.Paint;

/**
 * Prosedürel görsel üretici. Gerçek AI DEĞİL — algoritmik.
 * Kullanıcı prompt'undan renk paleti + geometrik kompozisyon üretir.
 * SR ile birleştirilerek "büyütülmüş" çıktı verilir.
 */
public class ProceduralImageGen {
    public static Bitmap generate(String prompt, int size) {
        int h = hash(prompt);
        int baseHue = Math.abs(h) % 360;
        Bitmap bmp = Bitmap.createBitmap(size, size, Bitmap.Config.ARGB_8888);
        Canvas c = new Canvas(bmp);
        Paint p = new Paint();
        p.setAntiAlias(true);

        // Arka plan gradyan
        for (int y = 0; y < size; y++) {
            float t = (float) y / size;
            int col = Color.HSVToColor(new float[]{
                (baseHue + t * 40) % 360, 0.6f, 0.2f + t * 0.3f
            });
            p.setColor(col);
            c.drawRect(0, y, size, y + 1, p);
        }

        // Yıldız/şekil katmanı
        java.util.Random rng = new java.util.Random(h);
        int n = 6 + rng.nextInt(10);
        for (int i = 0; i < n; i++) {
            float[] hsv = { (baseHue + rng.nextInt(90)) % 360, 0.7f, 0.6f + rng.nextFloat() * 0.4f };
            p.setColor(Color.HSVToColor(hsv));
            p.setAlpha(60 + rng.nextInt(150));
            int r = size / 6 + rng.nextInt(size / 4);
            c.drawCircle(rng.nextInt(size), rng.nextInt(size), r, p);
        }

        // Ön plan çizgiler
        p.setColor(Color.HSVToColor(new float[]{ (baseHue + 180) % 360, 0.9f, 0.9f }));
        p.setStrokeWidth(2f);
        for (int i = 0; i < 8; i++) {
            c.drawLine(rng.nextInt(size), rng.nextInt(size), rng.nextInt(size), rng.nextInt(size), p);
        }
        return bmp;
    }

    private static int hash(String s) {
        int h = 0;
        for (char c : s.toCharArray()) h = h * 31 + c;
        return h;
    }
}
```

```java
package com.jai.app.video;

import android.graphics.Bitmap;
import android.graphics.Canvas;
import android.graphics.Color;
import android.graphics.Paint;

/**
 * Prosedürel video kare üretici.
 * Prompt'tan basit animasyon kareleri üretir; SR ile büyütülür.
 */
public class FrameGenerator {
    private final int width, height;
    private final int baseHue;
    private final int seed;

    public FrameGenerator(String prompt, int width, int height) {
        this.width = width;
        this.height = height;
        this.seed = prompt.hashCode();
        this.baseHue = Math.abs(seed) % 360;
    }

    public Bitmap renderFrame(int frameIndex, int totalFrames) {
        float t = (float) frameIndex / Math.max(1, totalFrames);
        Bitmap bmp = Bitmap.createBitmap(width, height, Bitmap.Config.ARGB_8888);
        Canvas c = new Canvas(bmp);
        Paint p = new Paint();
        p.setAntiAlias(true);

        // arka plan gradyan
        for (int y = 0; y < height; y++) {
            float yt = (float) y / height;
            int col = Color.HSVToColor(new float[]{
                (baseHue + yt * 30 + t * 60) % 360, 0.7f, 0.15f + yt * 0.2f
            });
            p.setColor(col);
            c.drawRect(0, y, width, y + 1, p);
        }

        // hareketli şekiller
        java.util.Random rng = new java.util.Random(seed);
        int n = 8;
        for (int i = 0; i < n; i++) {
            float phase = (t + rng.nextFloat()) % 1f;
            float cx = (float) (width * (0.5 + 0.4 * Math.sin(phase * 6.28 + i)));
            float cy = (float) (height * (0.5 + 0.4 * Math.cos(phase * 6.28 + i * 1.7)));
            int r = 20 + rng.nextInt(width / 8);
            p.setColor(Color.HSVToColor(new float[]{
                (baseHue + i * 30 + t * 120) % 360, 0.8f, 0.8f
            }));
            p.setAlpha(100 + rng.nextInt(120));
            c.drawCircle(cx, cy, r, p);
        }
        return bmp;
    }
}
```

```java
package com.jai.app.video;

import android.content.Context;
import android.graphics.Bitmap;
import com.jai.app.data.VideoJobEntity;
import com.jai.app.sr.SRPipeline;

import java.io.File;

public class VideoPipeline {
    private final Context ctx;
    private final SRPipeline sr;
    public VideoPipeline(Context ctx, SRPipeline sr) {
        this.ctx = ctx; this.sr = sr;
    }

    /** Checkpoint'ten devam edebilir. Yarıda kalırsa framesDone saklanır. */
    public boolean runJob(VideoJobEntity job, ProgressListener listener) {
        try {
            File dir = new File(ctx.getFilesDir(), "outputs/video/job_" + job.id);
            if (!dir.exists()) dir.mkdirs();

            int w = job.quality; // 512 veya 1024
            int h = job.quality;
            FrameGenerator gen = new FrameGenerator(job.prompt, w, h);
            VideoEncoder enc = new VideoEncoder(dir);

            for (int i = job.framesDone; i < job.framesTotal; i++) {
                Bitmap frame = gen.renderFrame(i, job.framesTotal);
                if (sr != null && w == 512) {
                    frame = sr.upscale(frame); // 1024'e büyüt
                }
                enc.writeFrame(frame, i);
                job.framesDone = i + 1;
                if (listener != null) listener.onProgress(i + 1, job.framesTotal);
                frame.recycle();
            }
            job.status = "DONE";
            job.outputPath = dir.getAbsolutePath();
            return true;
        } catch (Exception e) {
            job.status = "FAILED";
            job.note = e.getMessage();
            return false;
        }
    }

    public interface ProgressListener {
        void onProgress(int done, int total);
    }
}
```

```java
package com.jai.app.video;

import android.content.Context;
import androidx.annotation.NonNull;
import androidx.work.Worker;
import androidx.work.WorkerParameters;

import com.jai.app.data.AppDatabase;
import com.jai.app.data.VideoJobEntity;
import com.jai.app.sr.SRLoader;
import com.jai.app.sr.SRPipeline;

/**
 * WorkManager worker: kuyruktaki video işlerini sırayla işler.
 * Uygulama kapansa da devam eder (Android izin verdiği sürece).
 */
public class VideoWorker extends Worker {
    public VideoWorker(@NonNull Context ctx, @NonNull WorkerParameters p) {
        super(ctx, p);
    }

    @NonNull @Override
    public Result doWork() {
        AppDatabase db = AppDatabase.get(getApplicationContext());
        VideoJobEntity job = db.videoJobs().nextPending();
        if (job == null) return Result.success();

        job.status = "RUNNING";
        job.updatedAt = System.currentTimeMillis();
        db.videoJobs().update(job);

        SRPipeline sr = new SRPipeline(SRLoader.load(getApplicationContext()));
        VideoPipeline pipeline = new VideoPipeline(getApplicationContext(), sr);
        boolean ok = pipeline.runJob(job, (done, total) -> {
            job.framesDone = done;
            job.updatedAt = System.currentTimeMillis();
            db.videoJobs().update(job);
            setProgressAsync(new androidx.work.Data.Builder()
                    .putInt("done", done).putInt("total", total).build());
        });

        job.updatedAt = System.currentTimeMillis();
        db.videoJobs().update(job);
        return ok ? Result.success() : Result.retry();
    }
}
```

```java
package com.jai.app.brain;

public final class ExpressionEvaluator {
    public static Double eval(String raw) {
        if (raw == null) return null;
        String s = raw.replace(",", ".").replace(" ", "");
        if (s.isEmpty()) return null;
        try {
            Parser p = new Parser(s);
            double v = p.parseExpression();
            if (p.pos != s.length()) return null;
            if (Double.isNaN(v) || Double.isInfinite(v)) return null;
            return v;
        } catch (RuntimeException e) { return null; }
    }

    private static final class Parser {
        final String s; int pos = 0;
        Parser(String s) { this.s = s; }
        double parseExpression() {
            double v = parseTerm();
            while (pos < s.length() && (s.charAt(pos) == '+' || s.charAt(pos) == '-')) {
                char op = s.charAt(pos++);
                double r = parseTerm();
                v = (op == '+') ? v + r : v - r;
            }
            return v;
        }
        double parseTerm() {
            double v = parseFactor();
            while (pos < s.length() && (s.charAt(pos) == '*' || s.charAt(pos) == '/')) {
                char op = s.charAt(pos++);
                double r = parseFactor();
                if (op == '*') v *= r;
                else { if (r == 0) throw new ArithmeticException(); v /= r; }
            }
            return v;
        }
        double parseFactor() {
            if (pos < s.length() && s.charAt(pos) == '(') {
                pos++;
                double v = parseExpression();
                if (pos >= s.length() || s.charAt(pos) != ')') throw new RuntimeException();
                pos++;
                return v;
            }
            if (pos < s.length() && s.charAt(pos) == '-') { pos++; return -parseFactor(); }
            int start = pos;
            while (pos < s.length() && (Character.isDigit(s.charAt(pos)) || s.charAt(pos) == '.')) pos++;
            if (start == pos) throw new RuntimeException();
            return Double.parseDouble(s.substring(start, pos));
        }
    }
}
```

```java
    public ZipTool(Context ctx) { this.ctx = ctx; }

    public ToolResult execute(Map<String, Object> input) throws Exception {
        Object n = input.get("name");
        Object l = input.get("uris");
        if (n == null || l == null) return ToolResult.err("name/uris yok");
        List<String> uris = (List<String>) l;
        if (uris.isEmpty()) return ToolResult.err("uri listesi boş");
        File dir = new File(ctx.getFilesDir(), "outputs");
        if (!dir.exists() && !dir.mkdirs()) return ToolResult.err("dizin hatası");
        String zn = n.toString().endsWith(".zip") ? n.toString() : n + ".zip";
        File zf = new File(dir, zn);
        try (ZipOutputStream zos = new ZipOutputStream(new FileOutputStream(zf))) {
            int i = 0;

                    if (in != null) {
                        byte[] buf = new byte[8192]; int r;
                        while ((r = in.read(buf)) > 0) zos.write(buf, 0, r);
                    }
                }
                zos.closeEntry();
            }
        }
        Uri uri = FileProvider.getUriForFile(ctx, ctx.getPackageName() + ".files", zf);
```

```java
public class AuthManager {
    private final AppDatabase db;
    private final SharedPreferences prefs;

    public AuthManager(Context ctx, AppDatabase db) {
        this.db = db;
        this.prefs = ctx.getSharedPreferences("jai_prefs", Context.MODE_PRIVATE);
    }

    public static class RegisterResult {
        public final boolean ok;
        public final String message;
        public final User user;
        public RegisterResult(boolean ok, String msg, User u) {
            this.ok = ok; this.message = msg; this.user = u;
        }
    }

    public RegisterResult register(String name, String surname, String email, int age) {
        if (name == null || name.trim().isEmpty()) return new RegisterResult(false, "İsim boş olamaz", null);
        if (email == null || !email.contains("@")) return new RegisterResult(false, "Geçersiz e-posta", null);
        if (age < 1 || age > 120) return new RegisterResult(false, "Geçersiz yaş", null);

        String n = name.trim();
        String e = email.trim().toLowerCase();

        if (db.users().byName(n) != null)
            return new RegisterResult(false, "Bu isim zaten kayıtlı, farklı bir isim seç", null);
        if (db.users().byEmail(e) != null)
            return new RegisterResult(false, "Bu e-posta zaten kayıtlı, farklı bir e-posta kullan", null);

        String code = UUID.randomUUID().toString().substring(0, 8).toUpperCase();
        User u = new User(n, surname == null ? "" : surname.trim(), e, age, code);
        long id = db.users().insert(u);
        u.id = id;

        prefs.edit()
                .putLong("current_user_id", u.id)
                .putString("current_user_code", code)
                .apply();

        return new RegisterResult(true, "Kayıt başarılı. Kod: " + code, u);
    }

    public User currentUser() {
        long id = prefs.getLong("current_user_id", -1);
        if (id < 0) return null;
        for (User u : db.users().all()) if (u.id == id) return u;
        return null;
    }

    public String currentUserCode() {
        return prefs.getString("current_user_code", null);
    }

    public boolean isAdmin() {
        return prefs.getBoolean("is_admin", false);
    }

    public void setAdmin(boolean admin) {
        prefs.edit().putBoolean("is_admin", admin).apply();
    }

    public void logout() {
        prefs.edit().remove("current_user_id").remove("current_user_code").apply();
    }
}
```

```java
package com.jai.app.messaging;

import androidx.annotation.NonNull;
import androidx.annotation.Nullable;
import java.util.concurrent.TimeUnit;
import okhttp3.OkHttpClient;
import okhttp3.Request;
import okhttp3.Response;
import okhttp3.WebSocket;
import okhttp3.WebSocketListener;

/**
 * Mesajlaşma istemcisi. Sunucu URL yoksa bağlanmaz.
 * Durumlar gerçek sunucu yanıtlarına göre; sahte tik YOK.
 */
public class MessagingClient {
    public interface Listener {
        void onState(MessagingState s, String detail);
        void onMessage(String json);
    }
    private final OkHttpClient client = new OkHttpClient.Builder()
            .pingInterval(30, TimeUnit.SECONDS).build();
    private WebSocket ws;
    private MessagingState state = MessagingState.DISCONNECTED;
    @Nullable private Listener listener;
    private String serverUrl, userCode;

    public void setListener(@Nullable Listener l) { listener = l; }
    public MessagingState getState() { return state; }

    private void emit(MessagingState s, String d) {
        state = s;
        if (listener != null) listener.onState(s, d);
    }

    public void connect(String url, String code) {
        this.serverUrl = url;
        this.userCode = code;
        if (state == MessagingState.CONNECTED || state == MessagingState.CONNECTING) return;
        if (url == null || !url.startsWith("ws")) {
            emit(MessagingState.ERROR, "Geçersiz URL (ws:// / wss://)"); return;
        }
        emit(MessagingState.CONNECTING, url);
        Request req = new Request.Builder().url(url + "?user=" + code).build();
        ws = client.newWebSocket(req, new WebSocketListener() {
            @Override public void onOpen(@NonNull WebSocket w, @NonNull Response r) {
                emit(MessagingState.CONNECTED, "Bağlı");
            }
            @Override public void onMessage(@NonNull WebSocket w, @NonNull String text) {
                if (listener != null) listener.onMessage(text);
            }
            @Override public void onFailure(@NonNull WebSocket w, @NonNull Throwable t, @Nullable Response r) {
                emit(MessagingState.ERROR, "Hata: " + t.getMessage());
                scheduleReconnect();
            }
            @Override public void onClosed(@NonNull WebSocket w, int c, @NonNull String reason) {
                emit(MessagingState.DISCONNECTED, "Kapandı: " + reason);
            }
        });
    }

    private void scheduleReconnect() {
        if (serverUrl == null) return;
        new android.os.Handler(android.os.Looper.getMainLooper()).postDelayed(() -> {
            if (state != MessagingState.CONNECTED) {
                emit(MessagingState.CONNECTING, "Yeniden bağlanma...");
                connect(serverUrl, userCode);
            }
        }, 5000);
    }

    public boolean send(String json) {
        return ws != null && state == MessagingState.CONNECTED && ws.send(json);
    }
    public void disconnect() {
        if (ws != null) { ws.close(1000, "user"); ws = null; }
        emit(MessagingState.DISCONNECTED, "Kapandı");
    }
}
```

```java
package com.jai.server;

import org.java_websocket.WebSocket;
import org.java_websocket.handshake.ClientHandshake;
import org.java_websocket.server.WebSocketServer;

import java.net.InetSocketAddress;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

public class MessagingServer extends WebSocketServer {

    private final Map<String, WebSocket> online = new ConcurrentHashMap<>();
    private final Map<WebSocket, String> reverse = new ConcurrentHashMap<>();
    private UserStore users;
    private MessageStore msgs;

    public MessagingServer(int port, String dbPath) throws Exception {
        super(new InetSocketAddress(port));
        users = new UserStore(dbPath);
        msgs = new MessageStore(dbPath);
    }

    @Override public void onOpen(WebSocket c, ClientHandshake h) {
        String query = h.getResourceDescriptor();
        int i = query.indexOf("user=");
        if (i >= 0) {
            String code = query.substring(i + 5);
            int amp = code.indexOf('&');
            if (amp >= 0) code = code.substring(0, amp);
            online.put(code, c);
            reverse.put(c, code);
            System.out.println("[+] online: " + code);
        }
    }

    @Override public void onClose(WebSocket c, int code, String reason, boolean remote) {
        String user = reverse.remove(c);
        if (user != null) online.remove(user);
        System.out.println("[-] offline: " + user);
    }

    @Override public void onMessage(WebSocket c, String msg) {
        try {
            org.json.JSONObject j = new org.json.JSONObject(msg);
            String type = j.optString("type");
            if ("register".equals(type)) {
                String code = users.register(j.optString("display", "user"));
                c.send(new org.json.JSONObject()
                        .put("type", "registered").put("code", code).toString());
            } else if ("msg".equals(type)) {
                String from = reverse.get(c);
                String to = j.getString("to");
                String body = j.getString("body");
                long cid = j.optLong("cid", 0);
                long id = msgs.add(from, to, body);
                c.send(new org.json.JSONObject()
                        .put("type", "ack").put("id", id).put("cid", cid).toString());
                WebSocket target = online.get(to);
                if (target != null && target.isOpen()) {
                    target.send(new org.json.JSONObject()
                            .put("type", "msg").put("id", id)
                            .put("from", from).put("body", body).toString());
                }
            } else if ("delivered".equals(type)) {
                long id = j.getLong("id");
                msgs.markDelivered(id);
                String from = msgs.fromOf(id);
                WebSocket sender = from != null ? online.get(from) : null;
                if (sender != null && sender.isOpen()) {
                    sender.send(new org.json.JSONObject()
                            .put("type", "status").put("id", id).put("state", "DELIVERED").toString());
                }
            } else if ("read".equals(type)) {
                long id = j.getLong("id");
                msgs.markRead(id);
                String from = msgs.fromOf(id);
                WebSocket sender = from != null ? online.get(from) : null;
                if (sender != null && sender.isOpen()) {
                    sender.send(new org.json.JSONObject()
                            .put("type", "status").put("id", id).put("state", "READ").toString());
                }
            }
        } catch (Exception e) {
            System.err.println("mesaj hatası: " + e.getMessage());
        }
    }

    @Override public void onError(WebSocket c, Exception ex) {
        System.err.println("ws hata: " + ex.getMessage());
    }

    @Override public void onStart() {
        System.out.println("Ja.i sunucu port " + getPort());
    }

    public static void main(String[] args) throws Exception {
        int port = args.length > 0 ? Integer.parseInt(args[0]) : 8887;
        String db = args.length > 1 ? args[1] : "jai-server.db";
        new MessagingServer(port, db).start();
    }
}
```

```java
package com.jai.app.codelab;

/**
 * STATİK analiz motoru. Kod ÇALIŞTIRMAZ.
 * Sadece sözdizimi dengesi ve JSON geçerliliği kontrolü yapar.
 */
public class CodeLabEngine {
    public static class Analysis {
        public final boolean ok; public final String report;
        public Analysis(boolean ok, String r) { this.ok = ok; report = r; }
    }
    public Analysis analyze(String code, String lang) {
        if (code == null || code.isEmpty()) return new Analysis(false, "Boş kod");
        String l = lang == null ? "txt" : lang.toLowerCase();
        StringBuilder rep = new StringBuilder();

        if (l.equals("java") || l.equals("javascript") || l.equals("js")) {
            int braces = 0, parens = 0, brackets = 0;
            boolean lineC = false, blockC = false, str = false, chr = false;
            char prev = 0;
            for (int i = 0; i < code.length(); i++) {
                char c = code.charAt(i);
                if (lineC) { if (c == '\n') lineC = false; prev = c; continue; }
                if (blockC) { if (prev == '*' && c == '/') blockC = false; prev = c; continue; }
                if (!str && !chr) {
                    if (prev == '/' && c == '/') { lineC = true; prev = c; continue; }
                    if (prev == '/' && c == '*') { blockC = true; prev = c; continue; }
                }
                if (c == '"' && prev != '\\') str = !str;
                if (c == '\'' && prev != '\\') chr = !chr;
                if (!str && !chr) {
                    if (c == '{') braces++;
                    else if (c == '}') braces--;
                    else if (c == '(') parens++;
                    else if (c == ')') parens--;
                    else if (c == '[') brackets++;
                    else if (c == ']') brackets--;
                }
                prev = c;
            }
            if (braces != 0) rep.append("✗ {} dengesi bozuk (").append(braces).append(")\n");
            if (parens != 0) rep.append("✗ () dengesi bozuk (").append(parens).append(")\n");
            if (brackets != 0) rep.append("✗ [] dengesi bozuk (").append(brackets).append(")\n");
            if (str) rep.append("✗ Kapanmamış string\n");
        }
        if (l.equals("json")) {
            try { new org.json.JSONObject(code); rep.append("✓ JSON geçerli\n"); }
            catch (Exception e1) {
                try { new org.json.JSONArray(code); rep.append("✓ JSON dizi geçerli\n"); }
                catch (Exception e2) { rep.append("✗ JSON geçersiz: ").append(e2.getMessage()).append('\n'); }
            }
        }
        if (rep.length() == 0) rep.append("✓ Sözdizimi temel kontrolü geçti (tam derleme DEĞİL)\n");
        return new Analysis(!rep.toString().contains("✗"), rep.toString());
    }
}
```
