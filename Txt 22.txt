package com.jai.app.messaging;
import androidx.annotation.NonNull; import androidx.annotation.Nullable; import okhttp3.*; import java.util.concurrent.TimeUnit;
public class MessagingClient {
    public interface Listener { void onState(MessagingState state, String detail); void onMessage(String json); }
    private WebSocket ws; private final OkHttpClient client = new OkHttpClient.Builder().pingInterval(30, TimeUnit.SECONDS).build();
    private MessagingState state = MessagingState.DISCONNECTED; private Listener listener;
    public void setListener(Listener l) { this.listener = l; }
    public void connect(String url, String user) {
        if (state == MessagingState.CONNECTED) return; state = MessagingState.CONNECTING; if (listener != null) listener.onState(state, "Bağlanıyor...");
        ws = client.newWebSocket(new Request.Builder().url(url + "?user=" + user).build(), new WebSocketListener() {
            @Override public void onOpen(@NonNull WebSocket w, @NonNull Response r) { state = MessagingState.CONNECTED; if (listener != null) listener.onState(state, "Bağlandı"); }
            @Override public void onMessage(@NonNull WebSocket w, @NonNull String text) { if (listener != null) listener.onMessage(text); }
            @Override public void onFailure(@NonNull WebSocket w, @NonNull Throwable t, @Nullable Response r) { state = MessagingState.ERROR; if (listener != null) listener.onState(state, "Hata: " + t.getMessage()); }
            @Override public void onClosed(@NonNull WebSocket w, int code, @NonNull String reason) { state = MessagingState.DISCONNECTED; if (listener != null) listener.onState(state, "Kapandı"); }
        });
    }
    public void send(String json) { if (ws != null && state == MessagingState.CONNECTED) ws.send(json); }
    public void close() { if (ws != null) { ws.close(1000, "Bye"); ws = null; } state = MessagingState.DISCONNECTED; }
}

package com.jai.app.video;
import android.graphics.Bitmap; import android.media.MediaCodec; import android.media.MediaCodecInfo; import android.media.MediaFormat; import android.media.MediaMuxer; import java.io.File; import java.nio.ByteBuffer;
public class VideoEncoder {
    private MediaCodec videoCodec; private MediaCodec audioCodec; private MediaMuxer muxer;
    private int videoTrack = -1; private int audioTrack = -1; private boolean muxerStarted = false; private long frameTimeUs = 0;
    public VideoEncoder(File dir, int w, int h) throws Exception {
        MediaFormat vFmt = MediaFormat.createVideoFormat("video/avc", w, h);
        vFmt.setInteger(MediaFormat.KEY_COLOR_FORMAT, MediaCodecInfo.CodecCapabilities.COLOR_FormatYUV420SemiPlanar);
        vFmt.setInteger(MediaFormat.KEY_BIT_RATE, 4000000); vFmt.setInteger(MediaFormat.KEY_FRAME_RATE, 30); vFmt.setInteger(MediaFormat.KEY_I_FRAME_INTERVAL, 1);
        videoCodec = MediaCodec.createEncoderByType("video/avc"); videoCodec.configure(vFmt, null, null, MediaCodec.CONFIGURE_FLAG_ENCODE); videoCodec.start();
        MediaFormat aFmt = MediaFormat.createAudioFormat("audio/mp4a-latm", 44100, 2);
        aFmt.setInteger(MediaFormat.KEY_BIT_RATE, 64000); aFmt.setInteger(MediaFormat.KEY_AAC_PROFILE, MediaCodecInfo.CodecProfileLevel.AACObjectLC);
        audioCodec = MediaCodec.createEncoderByType("audio/mp4a-latm"); audioCodec.configure(aFmt, null, null, MediaCodec.CONFIGURE_FLAG_ENCODE); audioCodec.start();
        muxer = new MediaMuxer(new File(dir, "video.mp4").getAbsolutePath(), MediaMuxer.OutputFormat.MUXER_OUTPUT_MPEG_4);
    }
    public void writeFrame(Bitmap bmp, int idx) throws Exception {
        int inIdx = videoCodec.dequeueInputBuffer(10000);
        if (inIdx >= 0) {
            ByteBuffer buf = videoCodec.getInputBuffer(inIdx);
            if (buf != null) {
                int[] pixels = new int[bmp.getWidth() * bmp.getHeight()]; bmp.getPixels(pixels, 0, bmp.getWidth(), 0, 0, bmp.getWidth(), bmp.getHeight());
                byte[] yuv = rgbToYuv420SemiPlanar(pixels, bmp.getWidth(), bmp.getHeight()); buf.clear(); buf.put(yuv);
                videoCodec.queueInputBuffer(inIdx, 0, yuv.length, frameTimeUs, 0);
            }
        }
        frameTimeUs += 33333L;
        int aInIdx = audioCodec.dequeueInputBuffer(10000);
        if (aInIdx >= 0) {
            ByteBuffer buf = audioCodec.getInputBuffer(aInIdx);
            if (buf != null) { byte[] silence = new byte[2048]; buf.clear(); buf.put(silence); audioCodec.queueInputBuffer(aInIdx, 0, silence.length, frameTimeUs - 33333L, 0); }
        }
        drainCodec(videoCodec, true); drainCodec(audioCodec, false);
    }
    private void drainCodec(MediaCodec codec, boolean isVideo) {
        MediaCodec.BufferInfo info = new MediaCodec.BufferInfo(); int outIdx = codec.dequeueOutputBuffer(info, 0);
        if (outIdx == MediaCodec.INFO_OUTPUT_FORMAT_CHANGED) {
            if (isVideo) videoTrack = muxer.addTrack(codec.getOutputFormat()); else audioTrack = muxer.addTrack(codec.getOutputFormat());
            if (videoTrack != -1 && audioTrack != -1 && !muxerStarted) { muxer.start(); muxerStarted = true; }
        } else if (outIdx >= 0) {
            ByteBuffer outBuf = codec.getOutputBuffer(outIdx);
            if (outBuf != null && info.size > 0 && muxerStarted) {
                outBuf.position(info.offset); outBuf.limit(info.offset + info.size); muxer.writeSampleData(isVideo ? videoTrack : audioTrack, outBuf, info);
            }
            codec.releaseOutputBuffer(outIdx, false);
        }
    }
    private byte[] rgbToYuv420SemiPlanar(int[] argb, int width, int height) {
        int frameSize = width * height; byte[] yuv = new byte[frameSize * 3 / 2]; int yIndex = 0, uvIndex = frameSize;
        for (int j = 0; j < height; j++) {
            for (int i = 0; i < width; i++) {
                int r = (argb[j * width + i] >> 16) & 0xff; int g = (argb[j * width + i] >> 8) & 0xff; int b = argb[j * width + i] & 0xff;
                int y = ((66 * r + 129 * g + 25 * b + 128) >> 8) + 16;
                yuv[yIndex++] = (byte) Math.max(0, Math.min(255, y));
                if (j % 2 == 0 && i % 2 == 0) {
                    int u = ((-38 * r - 74 * g + 112 * b + 128) >> 8) + 128; int v = ((112 * r - 94 * g - 18 * b + 128) >> 8) + 128;
                    yuv[uvIndex++] = (byte) Math.max(0, Math.min(255, u)); yuv[uvIndex++] = (byte) Math.max(0, Math.min(255, v));
                }
            }
        }
        return yuv;
    }
    public void finish() {
        if (videoCodec != null) { videoCodec.stop(); videoCodec.release(); }
        if (audioCodec != null) { audioCodec.stop(); audioCodec.release(); }
        if (muxer != null && muxerStarted) { muxer.stop(); muxer.release(); }
    }
}

            int inIdx = codec.dequeueInputBuffer(1000);

                codec.releaseOutputBuffer(outIdx, false);
            }
        } catch (Exception ignored) {}
    }

public class SRModel {
    public int[] upscale(int[] in, int w, int h) {
        int ow = w * 2, oh = h * 2; int[] out = new int[ow * oh];
        for (int y = 0; y < oh; y++) for (int x = 0; x < ow; x++) out[y * ow + x] = in[(y / 2) * w + (x / 2)];
        return out;
    }
}

            }
            return count == 0 ? ToolResult.ok("(Sonuç yok)") : ToolResult.ok(sb.toString());

package com.jai.app.codelab;
public class CodeLabEngine {
    public static class Analysis { public final boolean ok; public final String report; public Analysis(boolean ok, String report) { this.ok = ok; this.report = report; } }
    public Analysis analyze(String code, String lang) {
        if (code == null || code.isEmpty()) return new Analysis(false, "Boş kod");
        if (lang.equals("json")) { try { new org.json.JSONObject(code); return new Analysis(true, "Geçerli JSON"); } catch (Exception e) { return new Analysis(false, "Geçersiz JSON: " + e.getMessage()); } }
        int braces = 0, parens = 0; boolean inString = false, inChar = false, inLineComment = false, inBlockComment = false; char prev = 0;
        for (char c : code.toCharArray()) {
            if (inLineComment) { if (c == '\n') inLineComment = false; continue; }
            if (inBlockComment) { if (prev == '*' && c == '/') inBlockComment = false; prev = c; continue; }
            if (!inString && !inChar) { if (prev == '/' && c == '/') { inLineComment = true; prev = c; continue; } if (prev == '/' && c == '*') { inBlockComment = true; prev = c; continue; } }
            if (c == '"' && prev != '\\') inString = !inString; if (c == '\'' && prev != '\\') inChar = !inChar;
            if (!inString && !inChar) { if (c == '{') braces++; else if (c == '}') braces--; else if (c == '(') parens++; else if (c == ')') parens--; }
            prev = c;
        }
        if (braces == 0 && parens == 0 && !inString) return new Analysis(true, "Denge tamam");
        return new Analysis(false, "Denge bozuk: {}=" + braces + " ()=" + parens + " string=" + inString);
    }
}

import com.jai.app.data.KnowledgeEntity;
public class LearningVerifier {
    public boolean conflicts(KnowledgeEntity existing, String newContent) {
        if (existing == null || newContent == null) return false;
        String a = existing.content.toLowerCase().replaceAll("[^a-z0-9ğüşıöç ]", "");
        String b = newContent.toLowerCase().replaceAll("[^a-z0-9ğüşıöç ]", "");
        String[] wordsA = a.split("\\s+"); String[] wordsB = b.split("\\s+");
        if (wordsA.length < 3 || wordsB.length < 3) return false;
        int common = 0; for (String wa : wordsA) for (String wb : wordsB) if (wa.equals(wb) && wa.length() > 3) { common++; break; }
        return common > 2 && (b.contains("değil") || b.contains("yanlış"));
    }
}

package com.jai.app.trainer;
import com.jai.app.model.Matrix;
public class Loss {
    public static float crossEntropy(Matrix logits, int[] targets) {
        float total = 0;
        for (int i = 0; i < logits.rows; i++) {
            int row = i * logits.cols; float max = Float.NEGATIVE_INFINITY;
            for (int j = 0; j < logits.cols; j++) if (logits.data[row + j] > max) max = logits.data[row + j];
            float sum = 0; for (int j = 0; j < logits.cols; j++) sum += (float) Math.exp(logits.data[row + j] - max);
            total -= (logits.data[row + targets[i]] - max - (float) Math.log(sum));
        }
        return total / logits.rows;
    }
}

package com.jai.app.trainer;
import com.jai.app.model.Matrix; import com.jai.app.model.TransformerModel; import java.io.DataInputStream; import java.io.InputStream;
public class PtqReader {
    public static TransformerModel read(InputStream in) throws Exception {
        DataInputStream din = new DataInputStream(in); int vocab = din.readInt(); int dim = din.readInt(); int layers = din.readInt();
        TransformerModel m = new TransformerModel(vocab, dim, layers, 8, 2304, 1024);
        readMatrix(din, m.tokenEmb.W); readMatrix(din, m.posEmb.W);
        for (var b : m.blocks) {
            readMatrix(din, b.attn.qProj.W); readMatrix(din, b.attn.kProj.W); readMatrix(din, b.attn.vProj.W); readMatrix(din, b.attn.oProj.W);
            readMatrix(din, b.ffn.fc1.W); readMatrix(din, b.ffn.fc2.W);
        }
        readMatrix(din, m.lmHead.W); return m;
    }
    private static void readMatrix(DataInputStream din, Matrix target) throws Exception {
        int rows = din.readInt(); int cols = din.readInt(); float scale = din.readFloat(); int zp = din.readInt();
        byte[] data = new byte[rows * cols]; din.readFully(data);
        for (int i = 0; i < data.length; i++) target.data[i] = ((data[i] & 0xFF) - zp) * scale;
    }
}

public class PositionalEncoding {
    public static Matrix encode(int seqLen, int dim) {
        Matrix pos = new Matrix(seqLen, dim);
        for (int i = 0; i < seqLen; i++) {
            for (int j = 0; j < dim; j += 2) {
                double angle = i / Math.pow(10000, j / (double) dim);
                pos.data[i * dim + j] = (float) Math.sin(angle);
                if (j + 1 < dim) pos.data[i * dim + j + 1] = (float) Math.cos(angle);
            }
        }
        return pos;
    }
}

import org.java_websocket.WebSocket; import org.java_websocket.handshake.ClientHandshake; import org.java_websocket.server.WebSocketServer;
import java.net.InetSocketAddress; import java.util.Map; import java.util.concurrent.ConcurrentHashMap;
public class MessagingServer extends WebSocketServer {
    private final Map<String, WebSocket> online = new ConcurrentHashMap<>();
    public MessagingServer(int port) { super(new InetSocketAddress(port)); }
    @Override public void onOpen(WebSocket c, ClientHandshake h) { String q = h.getResourceDescriptor(); int i = q.indexOf("user="); if (i >= 0) online.put(q.substring(i + 5), c); }
    @Override public void onClose(WebSocket c, int code, String reason, boolean remote) { online.values().remove(c); }
    @Override public void onMessage(WebSocket c, String msg) {
        try {
            org.json.JSONObject j = new org.json.JSONObject(msg);
            if (j.has("to")) { WebSocket t = online.get(j.getString("to")); if (t != null && t.isOpen()) t.send(msg); }
        } catch (Exception ignored) {}
    }
    @Override public void onError(WebSocket c, Exception ex) {}
    @Override public void onStart() {}
    public static void main(String[] a) { new MessagingServer(8887).start(); }
}

// DOSYA 145: app/src/main/java/com/jai/app/model/image/Conv2D.java
package com.jai.app.model.image;
import com.jai.app.model.Matrix;
public class Conv2D {
    public final int inChannels, outChannels, kernelSize, stride;
    public float[][][][] weights; public float[] bias;
    public Conv2D(int inChannels, int outChannels, int kernelSize, int stride) {
        this.inChannels = inChannels; this.outChannels = outChannels; this.kernelSize = kernelSize; this.stride = stride;
        this.weights = new float[outChannels][inChannels][kernelSize][kernelSize]; this.bias = new float[outChannels];
        java.util.Random r = new java.util.Random(42); float std = (float) Math.sqrt(2.0 / (inChannels * kernelSize * kernelSize));
        for (int oc = 0; oc < outChannels; oc++) for (int ic = 0; ic < inChannels; ic++) for (int ky = 0; ky < kernelSize; ky++) for (int kx = 0; kx < kernelSize; kx++) weights[oc][ic][ky][kx] = (float) (r.nextGaussian() * std);
    }
    public Matrix forward(Matrix input, int w, int h) {
        int outW = (w - kernelSize) / stride + 1; int outH = (h - kernelSize) / stride + 1;
        Matrix output = new Matrix(outW * outH, outChannels);
        for (int oc = 0; oc < outChannels; oc++) {
            for (int oy = 0; oy < outH; oy++) {
                for (int ox = 0; ox < outW; ox++) {
                    float sum = bias[oc];
                    for (int ic = 0; ic < inChannels; ic++) {
                        for (int ky = 0; ky < kernelSize; ky++) {
                            for (int kx = 0; kx < kernelSize; kx++) {
                                int iy = oy * stride + ky; int ix = ox * stride + kx;
                                if (iy < h && ix < w) { int inIdx = iy * w + ix; sum += input.data[inIdx * inChannels + ic] * weights[oc][ic][ky][kx]; }
                            }
                        }
                    }
                    output.data[(oy * outW + ox) * outChannels + oc] = sum;
                }
            }
        }
        return output;
    }
}

// DOSYA 159: app/src/main/java/com/jai/app/model/TransformerModel.java
package com.jai.app.model;
public class TransformerModel {
    public final int vocabSize, dim, numLayers, numHeads, ffnHidden, maxSeq;
    public final Embedding tokenEmb, posEmb; public final TransformerBlock[] blocks; public final LayerNorm finalNorm; public final Dense lmHead;
    public TransformerModel(int vocabSize, int dim, int numLayers, int numHeads, int ffnHidden, int maxSeq) {
        this.vocabSize = vocabSize; this.dim = dim; this.numLayers = numLayers; this.numHeads = numHeads; this.ffnHidden = ffnHidden; this.maxSeq = maxSeq;
        this.tokenEmb = new Embedding(vocabSize, dim); this.posEmb = new Embedding(maxSeq, dim);
        this.blocks = new TransformerBlock[numLayers];
        for (int i = 0; i < numLayers; i++) blocks[i] = new TransformerBlock(dim, numHeads, ffnHidden);
        this.finalNorm = new LayerNorm(dim); this.lmHead = new Dense(dim, vocabSize);
    }
    public Matrix forward(int[] tokenIds) {
        int seq = Math.min(tokenIds.length, maxSeq); Matrix X = tokenEmb.forward(tokenIds);
        int[] positions = new int[seq]; for (int i = 0; i < seq; i++) positions[i] = i;
        X = X.add(posEmb.forward(positions));
        for (TransformerBlock b : blocks) X = b.forward(X);
        X = finalNorm.forward(X); return lmHead.forward(X);
    }
}

package com.jai.app.ui;
import android.content.Intent; import android.os.Bundle; import android.os.Handler; import android.os.Looper; import android.widget.EditText; import android.widget.TextView;

import java.util.List; import java.util.concurrent.ExecutorService; import java.util.concurrent.Executors;
public class MainActivity extends AppCompatActivity {
    private final ExecutorService io = Executors.newSingleThreadExecutor(); private final Handler ui = new Handler(Looper.getMainLooper());
    private AppDatabase db; private Conversation conv; private MessageAdapter adapter; private BrainEngine brain; private ToolRegistry tools;
    @Override protected void onCreate(Bundle b) {
        super.onCreate(b); setContentView(R.layout.activity_main);
        db = AppDatabase.get(this); brain = new BrainEngine(JaiApp.get().runtime(), new RagEngine(db)); tools = new ToolRegistry(this, db);
        RecyclerView rv = findViewById(R.id.list); adapter = new MessageAdapter(); rv.setLayoutManager(new LinearLayoutManager(this)); rv.setAdapter(adapter);
        TextView state = findViewById(R.id.txtModelState);
        io.execute(() -> {
            while (JaiApp.get().runtime().getState().equals("LOADING")) { try { Thread.sleep(500); } catch (Exception ignored) {} }
            ui.post(() -> state.setText(JaiApp.get().runtime().getState()));
        });
        EditText input = findViewById(R.id.input);
        io.execute(() -> {
            List<Conversation> all = db.conversations().all();
            if (all.isEmpty()) { long id = db.conversations().insert(new Conversation("Sohbet")); conv = db.conversations().byId(id); }
            else conv = all.get(0);
            List<Message> msgs = db.messages().forConversation(conv.id); ui.post(() -> adapter.setItems(msgs));
        });
        findViewById(R.id.btnSend).setOnClickListener(v -> {
            String text = input.getText().toString().trim(); if (text.isEmpty() || conv == null) return;
            input.setText("");
            Message uMsg = new Message(conv.id, "user", text, "FLASH", "MID", false, "OK", null);
            adapter.append(uMsg); ui.post(() -> rv.scrollToPosition(adapter.getItemCount() - 1));
            io.execute(() -> {
                db.messages().insert(uMsg); db.conversations().touch(conv.id, System.currentTimeMillis());
                BrainEngine.Result res = brain.respond(text, "FLASH", "MID", false, tools);
                if (res.reply.startsWith("__NEEDS_LEARNING__")) {
                    String topic = res.reply.replace("__NEEDS_LEARNING__", "");
                    db.learningTasks().insert(new com.jai.app.data.LearningTaskEntity(topic, 10)); res.reply = "Öğrenme kuyruğuna eklendi: " + topic;
                } else if (res.reply.startsWith("__NEEDS_IMAGE__")) {
                    String prompt = res.reply.replace("__NEEDS_IMAGE__", ""); res.reply = "Görsel oluşturuluyor: " + prompt + ". Bu işlem için Görsel Üretim ekranını kullanın.";
                }
                Message nMsg = new Message(conv.id, "nova", res.reply, "FLASH", "MID", false, res.status, res.trace);
                db.messages().insert(nMsg); db.conversations().touch(conv.id, System.currentTimeMillis());
                ui.post(() -> { adapter.append(nMsg); rv.scrollToPosition(adapter.getItemCount() - 1); });
            });
        });
        findViewById(R.id.btnMenu).setOnClickListener(v -> {
            String[] items = {"Yeni Sohbet", "Görsel Üretimi", "Görsel Galerisi", "Video Üretimi", "Video Galerisi", "Öğrenme Merkezi", "Ayarlar"};
            new androidx.appcompat.app.AlertDialog.Builder(this).setItems(items, (d, w) -> {

                else if (w == 3) startActivity(new Intent(this, VideoActivity.class));
                else if (w == 4) startActivity(new Intent(this, VideoGalleryActivity.class));
                else if (w == 5) startActivity(new Intent(this, LearningActivity.class));
                else if (w == 6) startActivity(new Intent(this, SettingsActivity.class));
            }).show();
        });
    }

package com.jai.app.model;
public class Matrix {
    public final int rows, cols;
    public final float[] data;
    public Matrix(int r, int c) { this.rows = r; this.cols = c; this.data = new float[r * c]; }
    public Matrix(int r, int c, float[] d) { this.rows = r; this.cols = c; this.data = d; }
    public float get(int r, int c) { return data[r * cols + c]; }
    public void set(int r, int c, float v) { data[r * cols + c] = v; }
    public static Matrix mul(Matrix A, Matrix B) {
        if (A.cols != B.rows) throw new IllegalArgumentException("Boyut uyumsuz");
        Matrix C = new Matrix(A.rows, B.cols);
        for (int i = 0; i < A.rows; i++) {
            for (int k = 0; k < A.cols; k++) {
                float a = A.data[i * A.cols + k];
                if (a == 0) continue;
                for (int j = 0; j < B.cols; j++) C.data[i * C.cols + j] += a * B.data[k * B.cols + j];
            }
        }
        return C;
    }
    public Matrix transpose() {
        Matrix T = new Matrix(cols, rows);
        for (int r = 0; r < rows; r++) for (int c = 0; c < cols; c++) T.data[c * rows + r] = data[r * cols + c];
        return T;
    }
    public Matrix add(Matrix B) { Matrix C = new Matrix(rows, cols); for (int i = 0; i < data.length; i++) C.data[i] = data[i] + B.data[i]; return C; }
    public static Matrix softmax(Matrix X) {
        Matrix Y = new Matrix(X.rows, X.cols);
        for (int i = 0; i < X.rows; i++) {
            int row = i * X.cols; float max = Float.NEGATIVE_INFINITY;
            for (int j = 0; j < X.cols; j++) if (X.data[row + j] > max) max = X.data[row + j];
            float sum = 0;
            for (int j = 0; j < X.cols; j++) { float e = (float) Math.exp(X.data[row + j] - max); Y.data[row + j] = e; sum += e; }
            for (int j = 0; j < X.cols; j++) Y.data[row + j] /= sum;
        }
        return Y;
    }
    public static Matrix layerNorm(Matrix X, float[] gamma, float[] beta, float eps) {
        Matrix Y = new Matrix(X.rows, X.cols);
        for (int i = 0; i < X.rows; i++) {
            int row = i * X.cols; float mean = 0;
            for (int j = 0; j < X.cols; j++) mean += X.data[row + j];
            mean /= X.cols; float var = 0;
            for (int j = 0; j < X.cols; j++) { float d = X.data[row + j] - mean; var += d * d; }
            var /= X.cols; float inv = (float) (1.0 / Math.sqrt(var + eps));
            for (int j = 0; j < X.cols; j++) Y.data[row + j] = (X.data[row + j] - mean) * inv * gamma[j] + beta[j];
        }
        return Y;
    }
    public static Matrix gelu(Matrix X) {
        Matrix Y = new Matrix(X.rows, X.cols);
        for (int i = 0; i < X.data.length; i++) { float x = X.data[i]; Y.data[i] = 0.5f * x * (1f + (float) Math.tanh(0.7978845608f * (x + 0.044715f * x * x * x))); }
        return Y;
    }
    public static Matrix relu(Matrix X) { Matrix Y = new Matrix(X.rows, X.cols); for (int i = 0; i < X.data.length; i++) Y.data[i] = Math.max(0, X.data[i]); return Y; }
}

    public static Int8Tensor quantize(Matrix m) {
        float min = Float.MAX_VALUE, max = -Float.MAX_VALUE;
        for (float v : m.data) { if (v < min) min = v; if (v > max) max = v; }
        float scale = (max - min) / 255f; if (scale == 0) scale = 1e-8f;
        int zp = Math.round(-min / scale); if (zp < 0) zp = 0; if (zp > 255) zp = 255;

        return new Int8Tensor(m.rows, m.cols, q, scale, zp);
    }

package com.jai.app.model;
public class MultiHeadAttention {
    public final int dim, numHeads, headDim; public final Dense qProj, kProj, vProj, oProj;
    public MultiHeadAttention(int dim, int numHeads) { this.dim = dim; this.numHeads = numHeads; this.headDim = dim / numHeads; this.qProj = new Dense(dim, dim); this.kProj = new Dense(dim, dim); this.vProj = new Dense(dim, dim); this.oProj = new Dense(dim, dim); }
    public Matrix forward(Matrix X) {
        int seq = X.rows; Matrix Q = qProj.forward(X); Matrix K = kProj.forward(X); Matrix V = vProj.forward(X);
        float scale = (float) (1.0 / Math.sqrt(headDim)); Matrix out = new Matrix(seq, dim);
        for (int h = 0; h < numHeads; h++) {
            int off = h * headDim; Matrix scores = new Matrix(seq, seq);
            for (int i = 0; i < seq; i++) {
                for (int j = 0; j <= i; j++) { float s = 0; for (int d = 0; d < headDim; d++) s += Q.data[i * dim + off + d] * K.data[j * dim + off + d]; scores.data[i * seq + j] = s * scale; }
                for (int j = i + 1; j < seq; j++) scores.data[i * seq + j] = Float.NEGATIVE_INFINITY;
            }
            Matrix attn = Matrix.softmax(scores);
            for (int i = 0; i < seq; i++) { for (int d = 0; d < headDim; d++) { float s = 0; for (int j = 0; j <= i; j++) s += attn.data[i * seq + j] * V.data[j * dim + off + d]; out.data[i * dim + off + d] = s; } }
        }
        return oProj.forward(out);
    }
}

    public final int vocabSize, dim, numLayers, numHeads, ffnHidden, maxSeq;
    public final Embedding tokenEmb, posEmb; public final TransformerBlock[] blocks; public final LayerNorm finalNorm; public final Dense lmHead;
    public TransformerModel(int vocabSize, int dim, int numLayers, int numHeads, int ffnHidden, int maxSeq) {
        this.vocabSize = vocabSize; this.dim = dim; this.numLayers = numLayers; this.numHeads = numHeads; this.ffnHidden = ffnHidden; this.maxSeq = maxSeq;
        this.tokenEmb = new Embedding(vocabSize, dim); this.posEmb = new Embedding(maxSeq, dim);
        this.blocks = new TransformerBlock[numLayers];
        for (int i = 0; i < numLayers; i++) blocks[i] = new TransformerBlock(dim, numHeads, ffnHidden);
        this.finalNorm = new LayerNorm(dim); this.lmHead = new Dense(dim, vocabSize);
    }
    public Matrix forward(int[] tokenIds) {
        int seq = Math.min(tokenIds.length, maxSeq); Matrix X = tokenEmb.forward(tokenIds);
        int[] positions = new int[seq]; for (int i = 0; i < seq; i++) positions[i] = i;
        X = X.add(posEmb.forward(positions));
        for (TransformerBlock b : blocks) X = b.forward(X);
        X = finalNorm.forward(X); return lmHead.forward(X);
    }
}

import android.content.Context; import java.io.BufferedReader; import java.io.InputStreamReader; import java.util.ArrayList; import java.util.HashMap; import java.util.List; import java.util.Map; import org.json.JSONObject;
public class BpeTokenizer {
    private final Map<String, Integer> vocab = new HashMap<>(); private final Map<Integer, String> reverse = new HashMap<>(); private final List<String[]> merges = new ArrayList<>();
    public BpeTokenizer(Context ctx) {
        try {
            BufferedReader vr = new BufferedReader(new InputStreamReader(ctx.getAssets().open("vocab.json"))); StringBuilder sb = new StringBuilder(); String line;
            while ((line = vr.readLine()) != null) sb.append(line); JSONObject json = new JSONObject(sb.toString());
            for (String key : json.keySet()) { int id = json.getInt(key); vocab.put(key, id); reverse.put(id, key); } vr.close();
            BufferedReader mr = new BufferedReader(new InputStreamReader(ctx.getAssets().open("merges.txt"))); mr.readLine();
            while ((line = mr.readLine()) != null) { if (line.trim().isEmpty()) continue; merges.add(line.split(" ")); } mr.close();
        } catch (Exception e) { for (int i = 0; i < 256; i++) { String key = String.valueOf((char) i); vocab.put(key, i); reverse.put(i, key); } }
    }
    public int[] encode(String text) {
        byte[] bytes = text.getBytes(java.nio.charset.StandardCharsets.UTF_8); List<Integer> ids = new ArrayList<>();
        for (byte b : bytes) { String key = String.valueOf((char) (b & 0xFF)); Integer id = vocab.get(key); ids.add(id != null ? id : 0); }
        while (true) {
            int bestIdx = -1, bestRank = Integer.MAX_VALUE;
            for (int i = 0; i < ids.size() - 1; i++) {
                String p1 = reverse.get(ids.get(i)), p2 = reverse.get(ids.get(i+1));
                for (int rank = 0; rank < merges.size(); rank++) { if (merges.get(rank)[0].equals(p1) && merges.get(rank)[1].equals(p2)) { if (rank < bestRank) { bestRank = rank; bestIdx = i; } break; } }
            }
            if (bestIdx == -1) break; String merged = reverse.get(ids.get(bestIdx)) + reverse.get(ids.get(bestIdx + 1));
            Integer mergedId = vocab.get(merged); if (mergedId == null) break; ids.set(bestIdx, mergedId); ids.remove(bestIdx + 1);
        }
        int[] out = new int[ids.size()]; for (int i = 0; i < ids.size(); i++) out[i] = ids.get(i); return out;
    }
    public String decode(int[] ids) { StringBuilder sb = new StringBuilder(); for (int id : ids) { String tok = reverse.get(id); if (tok != null) sb.append(tok); } return sb.toString(); }
}
