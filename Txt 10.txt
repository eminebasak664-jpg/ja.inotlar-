    public final int vocabSize, dim;
    public Matrix W;
    public Embedding(int vocabSize, int dim) {
        this.vocabSize = vocabSize; this.dim = dim;
        this.W = new Matrix(vocabSize, dim);
        java.util.Random r = new java.util.Random(42);
        for (int i = 0; i < W.data.length; i++) W.data[i] = (float) (r.nextGaussian() * 0.02);
    }
    public Matrix forward(int[] tokenIds) {
        Matrix Y = new Matrix(tokenIds.length, dim);
        for (int i = 0; i < tokenIds.length; i++) {
            int id = tokenIds[i];
            if (id < 0 || id >= vocabSize) id = 0;
            System.arraycopy(W.data, id * dim, Y.data, i * dim, dim);
        }
        return Y;
    }
}

package com.jai.app.model;
public class MultiHeadAttention {
    public final int dim, numHeads, headDim;
    public final Dense qProj, kProj, vProj, oProj;
    public MultiHeadAttention(int dim, int numHeads) {
        this.dim = dim; this.numHeads = numHeads; this.headDim = dim / numHeads;
        this.qProj = new Dense(dim, dim); this.kProj = new Dense(dim, dim);
        this.vProj = new Dense(dim, dim); this.oProj = new Dense(dim, dim);
    }
    public Matrix forward(Matrix X) {
        int seq = X.rows; Matrix Q = qProj.forward(X); Matrix K = kProj.forward(X); Matrix V = vProj.forward(X);
        float scale = (float) (1.0 / Math.sqrt(headDim));
        Matrix out = new Matrix(seq, dim);
        for (int h = 0; h < numHeads; h++) {
            int off = h * headDim;
            Matrix scores = new Matrix(seq, seq);
            for (int i = 0; i < seq; i++) {
                for (int j = 0; j <= i; j++) {
                    float s = 0;
                    for (int d = 0; d < headDim; d++) s += Q.data[i * dim + off + d] * K.data[j * dim + off + d];
                    scores.data[i * seq + j] = s * scale;
                }
                for (int j = i + 1; j < seq; j++) scores.data[i * seq + j] = Float.NEGATIVE_INFINITY;
            }
            Matrix attn = Matrix.softmax(scores);
            for (int i = 0; i < seq; i++) {
                for (int d = 0; d < headDim; d++) {
                    float s = 0;
                    for (int j = 0; j <= i; j++) s += attn.data[i * seq + j] * V.data[j * dim + off + d];
                    out.data[i * dim + off + d] = s;
                }
            }
        }
        return oProj.forward(out);
    }
}

package com.jai.app.model;
public class TransformerModel {
    public final int vocabSize, dim, numLayers, numHeads, ffnHidden, maxSeq;
    public final Embedding tokenEmb, posEmb;
    public final TransformerBlock[] blocks;
    public final LayerNorm finalNorm;
    public final Dense lmHead;
    
    public TransformerModel(int vocabSize, int dim, int numLayers, int numHeads, int ffnHidden, int maxSeq) {
        this.vocabSize = vocabSize; this.dim = dim; this.numLayers = numLayers; this.numHeads = numHeads; this.ffnHidden = ffnHidden; this.maxSeq = maxSeq;
        this.tokenEmb = new Embedding(vocabSize, dim); this.posEmb = new Embedding(maxSeq, dim);
        this.blocks = new TransformerBlock[numLayers];
        for (int i = 0; i < numLayers; i++) blocks[i] = new TransformerBlock(dim, numHeads, ffnHidden);
        this.finalNorm = new LayerNorm(dim); this.lmHead = new Dense(dim, vocabSize);
    }
    
    public Matrix forward(int[] tokenIds) {
        int seq = Math.min(tokenIds.length, maxSeq);
        Matrix X = tokenEmb.forward(tokenIds);
        int[] positions = new int[seq];
        for (int i = 0; i < seq; i++) positions[i] = i;
        X = X.add(posEmb.forward(positions));
        for (TransformerBlock b : blocks) X = b.forward(X);
        X = finalNorm.forward(X);
        return lmHead.forward(X);
    }
}

package com.jai.app.model;
import android.content.Context;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.HashMap;
import java.util.Map;

public class BpeTokenizer {
    private final Map<String, Integer> vocab = new HashMap<>();
    private final Map<Integer, String> reverse = new HashMap<>();
    public BpeTokenizer(Context ctx) {
        try (BufferedReader r = new BufferedReader(new InputStreamReader(ctx.getAssets().open("vocab.txt")))) {
            String line; int i = 0;
            while ((line = r.readLine()) != null) { vocab.put(line, i); reverse.put(i, line); i++; }
        } catch (Exception e) {
            for (int i = 0; i < 256; i++) { vocab.put("b" + i, i); reverse.put(i, "b" + i); }
        }
    }
    public int[] encode(String text) {
        byte[] bytes = text.getBytes(java.nio.charset.StandardCharsets.UTF_8);
        int[] ids = new int[bytes.length];
        for (int i = 0; i < bytes.length; i++) ids[i] = vocab.getOrDefault("b" + (bytes[i] & 0xFF), 0);
        return ids;
    }
    public String decode(int[] ids) {
        StringBuilder sb = new StringBuilder();
        for (int id : ids) {
            String tok = reverse.get(id);
            if (tok != null && tok.startsWith("b")) { try { sb.append((char) Integer.parseInt(tok.substring(1))); } catch (Exception ignored) {} }
            else if (tok != null) sb.append(tok);
        }
        return sb.toString();
    }
}

package com.jai.app.model;
import java.util.Random;
public class Sampler {
    private final Random rng = new Random();
    public float temperature = 0.8f;
    public int topK = 40;
    
    public int sample(float[] logits) {
        int k = Math.min(topK, logits.length);
        Integer[] indices = new Integer[logits.length];
        for (int i = 0; i < logits.length; i++) indices[i] = i;
        java.util.Arrays.sort(indices, (a, b) -> Float.compare(logits[b], logits[a]));
        
        float[] probs = new float[k];
        float max = logits[indices[0]];
        float sum = 0;
        for (int i = 0; i < k; i++) {
            probs[i] = (float) Math.exp((logits[indices[i]] - max) / temperature);
            sum += probs[i];
        }
        for (int i = 0; i < k; i++) probs[i] /= sum;
        
        float r = rng.nextFloat();
        float cum = 0;
        for (int i = 0; i < k; i++) {
            cum += probs[i];
            if (r <= cum) return indices[i];
        }
        return indices[0];
    }
}

package com.jai.app.model;
import android.content.Context;
import android.content.res.AssetManager;
import com.jai.app.trainer.PtqReader;
import java.io.InputStream;

public class InferenceRuntime {
    public TransformerModel model;
    public BpeTokenizer tokenizer;
    private String state = "NOT_TRAINED";
    
    public void init(Context ctx) {
        tokenizer = new BpeTokenizer(ctx);
        AssetManager am = ctx.getAssets();
        try (InputStream in = am.open("model_int8.bin")) {
            model = PtqReader.read(in);
            state = "READY";
        } catch (Exception e) {
            model = new TransformerModel(16000, 576, 10, 8, 2304, 1024);
            state = "NOT_TRAINED";
        }
    }
    
    public String getState() { return state; }
    public boolean isReady() { return "READY".equals(state); }
    
    public String generate(String prompt, int maxTokens) {
        if (!isReady()) return "Model yüklenmedi: " + state;
        int[] ids = tokenizer.encode(prompt);
        StringBuilder out = new StringBuilder();
        Sampler sampler = new Sampler();
        int[] ctxIds = new int[Math.min(ids.length, model.maxSeq)];
        for (int i = 0; i < ctxIds.length; i++) ctxIds[i] = ids[ids.length - ctxIds.length + i];
        
        for (int step = 0; step < maxTokens; step++) {
            Matrix logits = model.forward(ctxIds);
            int last = logits.rows - 1;
            float[] lastLogits = new float[logits.cols];
            for (int i = 0; i < logits.cols; i++) lastLogits[i] = logits.data[last * logits.cols + i];
            
            int nextTok = sampler.sample(lastLogits);
            if (nextTok == 2) break; // EOS
            
            out.append(tokenizer.decode(new int[]{nextTok}));
            int[] newCtx = new int[Math.min(ctxIds.length + 1, model.maxSeq)];
            int skip = Math.max(0, ctxIds.length + 1 - model.maxSeq);
            for (int i = 0; i < ctxIds.length - skip; i++) newCtx[i] = ctxIds[i + skip];
            newCtx[newCtx.length - 1] = nextTok;
            ctxIds = newCtx;
        }
        return out.toString();
    }
}

// DOSYA 65: app/src/main/java/com/jai/app/brain/ExpressionEvaluator.java
package com.jai.app.brain;
public final class ExpressionEvaluator {
    public static Double eval(String raw) {
        if (raw == null) return null;
        String s = raw.replace(",", ".").replace(" ", "");
        if (s.isEmpty()) return null;
        try {
            Parser p = new Parser(s);
            double v = p.parseExpression();
            if (p.pos != s.length()) return null;
            if (Double.isNaN(v) || Double.isInfinite(v)) return null;
            return v;
        } catch (RuntimeException e) { return null; }
    }
    private static final class Parser {
        final String s; int pos = 0;
        Parser(String s) { this.s = s; }
        double parseExpression() {
            double v = parseTerm();
            while (pos < s.length() && (s.charAt(pos) == '+' || s.charAt(pos) == '-')) {
                char op = s.charAt(pos++); double r = parseTerm();
                v = (op == '+') ? v + r : v - r;
            }
            return v;
        }
        double parseTerm() {
            double v = parseFactor();
            while (pos < s.length() && (s.charAt(pos) == '*' || s.charAt(pos) == '/')) {
                char op = s.charAt(pos++); double r = parseFactor();
                if (op == '*') v *= r; else { if (r == 0) throw new ArithmeticException(); v /= r; }
            }
            return v;
        }
        double parseFactor() {
            if (pos < s.length() && s.charAt(pos) == '(') {
                pos++; double v = parseExpression();
                if (pos >= s.length() || s.charAt(pos) != ')') throw new RuntimeException();
                pos++; return v;
            }
            if (pos < s.length() && s.charAt(pos) == '-') { pos++; return -parseFactor(); }
            int start = pos;
            while (pos < s.length() && (Character.isDigit(s.charAt(pos)) || s.charAt(pos) == '.')) pos++;
            if (start == pos) throw new RuntimeException();
            return Double.parseDouble(s.substring(start, pos));
        }
    }
}

package com.jai.app.tools;
import android.content.Context;
import android.net.Uri;
import androidx.core.content.FileProvider;
import java.io.File;
import java.io.FileOutputStream;
import java.io.InputStream;
import java.util.List;
import java.util.Map;
import java.util.zip.ZipEntry;
import java.util.zip.ZipOutputStream;
public class ZipTool implements Tool {
    private final Context ctx;
    public ZipTool(Context ctx) { this.ctx = ctx; }
    public String id() { return "zip"; }
    public String name() { return "ZIP"; }
    @SuppressWarnings("unchecked")
    public ToolResult execute(Map<String, Object> input) throws Exception {
        List<String> uris = (List<String>) input.get("uris");
        if (uris == null || uris.isEmpty()) return ToolResult.err("uri yok");
        File dir = new File(ctx.getFilesDir(), "outputs"); dir.mkdirs();
        File zf = new File(dir, "archive_" + System.currentTimeMillis() + ".zip");
        try (ZipOutputStream zos = new ZipOutputStream(new FileOutputStream(zf))) {
            for (String u : uris) {
                String name = Uri.parse(u).getLastPathSegment();
                if (name == null) name = "file";
                name = name.replace("..", "_").replace("/", "_").replace("\\", "_");
                zos.putNextEntry(new ZipEntry(name));
                try (InputStream in = ctx.getContentResolver().openInputStream(Uri.parse(u))) {
                    if (in != null) { byte[] buf = new byte[8192]; int r; while ((r = in.read(buf)) > 0) zos.write(buf, 0, r); }
                }
                zos.closeEntry();
            }
        }
        return ToolResult.ok(FileProvider.getUriForFile(ctx, ctx.getPackageName() + ".files", zf).toString());
    }
}

package com.jai.app.auth;
import android.content.Context;
import android.content.SharedPreferences;
import com.jai.app.data.AppDatabase;
import com.jai.app.data.User;
import java.util.UUID;
public class AuthManager {
    private final AppDatabase db;
    private final SharedPreferences prefs;
    public AuthManager(Context ctx, AppDatabase db) {
        this.db = db; this.prefs = ctx.getSharedPreferences("jai_auth", Context.MODE_PRIVATE);
    }
    public String register(String name, String surname, String email, int age) {
        if (name == null || name.trim().isEmpty()) return "İsim boş olamaz";
        if (email == null || !email.contains("@")) return "Geçersiz e-posta";
        if (db.users().byName(name.trim()) != null) return "Bu isim zaten kayıtlı";
        if (db.users().byEmail(email.trim().toLowerCase()) != null) return "Bu e-posta zaten kayıtlı";
        User u = new User(name.trim(), surname, email.trim().toLowerCase(), age, UUID.randomUUID().toString().substring(0, 8).toUpperCase());
        long id = db.users().insert(u);
        prefs.edit().putLong("uid", id).apply();
        return "OK";
    }
    public User currentUser() {
        long id = prefs.getLong("uid", -1);
        if (id < 0) return null;
        for (User u : db.users().all()) if (u.id == id) return u;
        return null;
    }
    public boolean isAdmin() { return prefs.getBoolean("is_admin", false); }
    public void setAdmin(boolean admin) { prefs.edit().putBoolean("is_admin", admin).apply(); }
}
