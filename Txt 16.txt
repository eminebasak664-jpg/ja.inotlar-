package com.jai.app.data;
import androidx.room.Entity;
import androidx.room.PrimaryKey;

@Entity(tableName = "messages")
public class Message {
    @PrimaryKey(autoGenerate = true) public long id;

// DOSYA 44: app/src/main/java/com/jai/app/data/AppDatabase.java
package com.jai.app.data;
import android.content.Context;
import androidx.room.Database;
import androidx.room.Room;
import androidx.room.RoomDatabase;

@Database(entities = {Conversation.class, Message.class, User.class, KnowledgeEntity.class, LearningTaskEntity.class, VideoJobEntity.class}, version = 1, exportSchema = false)
public abstract class AppDatabase extends RoomDatabase {
    private static volatile AppDatabase INSTANCE;
    public abstract ConversationDao conversations();
    public abstract MessageDao messages();
    public abstract UserDao users();
    public abstract KnowledgeDao knowledge();
    public abstract LearningTaskDao learningTasks();
    public abstract VideoJobDao videoJobs();
    public static AppDatabase get(Context ctx) {
        if (INSTANCE == null) {
            synchronized (AppDatabase.class) {
                if (INSTANCE == null) {
                    INSTANCE = Room.databaseBuilder(ctx.getApplicationContext(), AppDatabase.class, "jai.db").fallbackToDestructiveMigration().build();
                }
            }
        }
        return INSTANCE;
    }
}
