        return ok ? Result.success() : Result.failure();
    }

package com.jai.app.messaging;
import androidx.annotation.NonNull;
import androidx.annotation.Nullable;
import okhttp3.*;
import java.util.concurrent.TimeUnit;

public class MessagingClient {
    public interface Listener {
        void onState(MessagingState state, String detail);
        void onMessage(String json);
    }
    private WebSocket ws; 
    private final OkHttpClient client = new OkHttpClient.Builder().pingInterval(30, TimeUnit.SECONDS).build();
    private MessagingState state = MessagingState.DISCONNECTED;
    private Listener listener;

    public void setListener(Listener l) { this.listener = l; }
    
    public void connect(String url, String user) {
        if (state == MessagingState.CONNECTED) return;
        state = MessagingState.CONNECTING;
        if (listener != null) listener.onState(state, "Bağlanıyor...");
        
        ws = client.newWebSocket(new Request.Builder().url(url + "?user=" + user).build(), new WebSocketListener() {
            @Override public void onOpen(@NonNull WebSocket w, @NonNull Response r) {
                state = MessagingState.CONNECTED;
                if (listener != null) listener.onState(state, "Bağlandı");
            }
            @Override public void onMessage(@NonNull WebSocket w, @NonNull String text) {
                if (listener != null) listener.onMessage(text);
            }
            @Override public void onFailure(@NonNull WebSocket w, @NonNull Throwable t, @Nullable Response r) {
                state = MessagingState.ERROR;
                if (listener != null) listener.onState(state, "Hata: " + t.getMessage());
            }
            @Override public void onClosed(@NonNull WebSocket w, int code, @NonNull String reason) {
                state = MessagingState.DISCONNECTED;
                if (listener != null) listener.onState(state, "Kapandı");
            }
        });
    }
    public void send(String json) { if (ws != null && state == MessagingState.CONNECTED) ws.send(json); }
    public void close() { if (ws != null) { ws.close(1000, "Bye"); ws = null; } state = MessagingState.DISCONNECTED; }
}

package com.jai.app.video;
import android.content.Context;
import android.graphics.Bitmap;
import com.jai.app.data.VideoJobEntity;
import com.jai.app.sr.SRPipeline;
import com.jai.app.sr.SRModel;
import java.io.File;

public class VideoPipeline {
    private final Context ctx;
    public VideoPipeline(Context ctx) { this.ctx = ctx; }
    public boolean runJob(VideoJobEntity job, ProgressListener listener) {
        try {
            File dir = new File(ctx.getFilesDir(), "outputs/video/job_" + job.id); dir.mkdirs();
            FrameGenerator gen = new FrameGenerator(job.prompt, job.quality, job.quality);
            VideoEncoder enc = new VideoEncoder(dir, job.quality, job.quality);
            SRPipeline sr = job.quality == 512 ? new SRPipeline(new SRModel()) : null;
            for (int i = job.framesDone; i < job.framesTotal; i++) {
                Bitmap f = gen.renderFrame(i, job.framesTotal);
                if (sr != null) f = sr.upscale(f);
                enc.writeFrame(f, i); f.recycle();
                if (listener != null) listener.onProgress(i + 1, job.framesTotal);
            }
            enc.finish();
            job.outputPath = new File(dir, "video.mp4").getAbsolutePath();
            return true;
        } catch (Exception e) { return false; }
    }
    public interface ProgressListener { void onProgress(int done, int total); }
}

// DOSYA 95: app/src/main/java/com/jai/app/video/VideoEncoder.java
package com.jai.app.video;
import android.graphics.Bitmap;
import android.media.MediaCodec;
import android.media.MediaCodecInfo;
import android.media.MediaFormat;
import android.media.MediaMuxer;
import java.io.File;
import java.nio.ByteBuffer;

public class VideoEncoder {
    private MediaCodec codec; private MediaMuxer muxer; private int track; private boolean muxerStarted;
    
    public VideoEncoder(File dir, int w, int h) throws Exception {
        MediaFormat fmt = MediaFormat.createVideoFormat("video/avc", w, h);
        fmt.setInteger(MediaFormat.KEY_COLOR_FORMAT, MediaCodecInfo.CodecCapabilities.COLOR_FormatYUV420Flexible);
        fmt.setInteger(MediaFormat.KEY_BIT_RATE, 4000000); fmt.setInteger(MediaFormat.KEY_FRAME_RATE, 24); fmt.setInteger(MediaFormat.KEY_I_FRAME_INTERVAL, 1);
        codec = MediaCodec.createEncoderByType("video/avc"); codec.configure(fmt, null, null, MediaCodec.CONFIGURE_FLAG_ENCODE); codec.start();
        muxer = new MediaMuxer(new File(dir, "video.mp4").getAbsolutePath(), MediaMuxer.OutputFormat.MUXER_OUTPUT_MPEG_4);
    }

    public void writeFrame(Bitmap bmp, int idx) throws Exception {
        MediaCodec.BufferInfo info = new MediaCodec.BufferInfo();
        int inIdx = codec.dequeueInputBuffer(10000);
        if (inIdx >= 0) {
            ByteBuffer buf = codec.getInputBuffer(inIdx);
            if (buf != null) {
                int[] pixels = new int[bmp.getWidth() * bmp.getHeight()]; bmp.getPixels(pixels, 0, bmp.getWidth(), 0, 0, bmp.getWidth(), bmp.getHeight());
                byte[] yuv = rgbToYuv(pixels, bmp.getWidth(), bmp.getHeight()); buf.clear(); buf.put(yuv);
                codec.queueInputBuffer(inIdx, 0, yuv.length, idx * 33333L, 0);
            }
        }
        int outIdx = codec.dequeueOutputBuffer(info, 10000);
        if (outIdx >= 0) {
            ByteBuffer outBuf = codec.getOutputBuffer(outIdx);
            if (outBuf != null && info.size > 0 && !muxerStarted) {
                MediaFormat outFmt = codec.getOutputFormat(); track = muxer.addTrack(outFmt); muxer.start(); muxerStarted = true;
            }
            if (muxerStarted && outBuf != null) { outBuf.position(info.offset); outBuf.limit(info.offset + info.size); muxer.writeSampleData(track, outBuf, info); }
            codec.releaseOutputBuffer(outIdx, false);
        }
    }

    public void finish() {
        if (codec != null) { codec.stop(); codec.release(); }
        if (muxer != null && muxerStarted) { muxer.stop(); muxer.release(); }
    }

    private byte[] rgbToYuv(int[] rgb, int w, int h) {
        byte[] yuv = new byte[w * h * 3 / 2];
        for (int i = 0; i < w * h; i++) { int r = (rgb[i] >> 16) & 0xff, g = (rgb[i] >> 8) & 0xff, b = rgb[i] & 0xff; yuv[i] = (byte) (0.299 * r + 0.587 * g + 0.114 * b); }
        return yuv;
    }
}

public class FrameGenerator {
    private final int w, h; private final int baseHue;
    public FrameGenerator(String prompt, int w, int h) { this.w = w; this.h = h; this.baseHue = Math.abs(prompt.hashCode()) % 360; }
    public Bitmap renderFrame(int idx, int total) {
        Bitmap bmp = Bitmap.createBitmap(w, h, Bitmap.Config.ARGB_8888); Canvas c = new Canvas(bmp); Paint p = new Paint();
        float t = (float) idx / total;
        c.drawColor(Color.HSVToColor(new float[]{ (baseHue + t * 60) % 360, 0.5f, 0.2f }));
        p.setColor(Color.HSVToColor(new float[]{ (baseHue + 180 + t * 90) % 360, 0.8f, 0.9f }));
        c.drawCircle(w/2f + (float)Math.sin(t*6.28)*w/4, h/2f + (float)Math.cos(t*6.28)*h/4, w/8f, p);
        return bmp;
    }
}

public class SRModel {
    public int[] upscale(int[] in, int w, int h) {
        int ow = w * 2, oh = h * 2; int[] out = new int[ow * oh];
        for (int y = 0; y < oh; y++) for (int x = 0; x < ow; x++) out[y * ow + x] = in[(y / 2) * w + (x / 2)];
        return out;
    }
}

package com.jai.app.image;
import android.content.Context;

import android.graphics.Color;

public class WebImageEngine {
    private final Context ctx;
    private final AppDatabase db;
    
    public WebImageEngine(Context ctx) {
        this.ctx = ctx;
        this.db = AppDatabase.get(ctx);
    }
    
    public Bitmap generate(String prompt) {
        // 1. Web'den bilgi topla (metin olarak)
        String context = "";
        try {
            var res = new WebResearchEngine(db).search(prompt + " nasıl görünür, rengi nedir");
            if (res.ok) context = res.output.toLowerCase();
        } catch (Exception ignored) {}
        
        // 2. Metinden renk ve şekil çıkar
        int mainColor = Color.GRAY;
        if (context.contains("kırmızı") || prompt.contains("kırmızı")) mainColor = Color.RED;
        else if (context.contains("mavi") || prompt.contains("mavi")) mainColor = Color.BLUE;
        else if (context.contains("yeşil") || prompt.contains("yeşil")) mainColor = Color.GREEN;
        else mainColor = Color.HSVToColor(new float[]{ Math.abs(prompt.hashCode()) % 360, 0.7f, 0.7f });
        
        // 3. Kendi çizim algoritmamızı çalıştır
        int size = 512;
        Bitmap bmp = Bitmap.createBitmap(size, size, Bitmap.Config.ARGB_8888);
        Canvas c = new Canvas(bmp);
        Paint p = new Paint();
        p.setAntiAlias(true);
        
        // Arka plan
        c.drawColor(Color.rgb(20, 20, 30));
        
        // Ana objeyi çiz (elma, araba vb. için basit vektörel temsil)
        p.setColor(mainColor);
        p.setStyle(Paint.Style.FILL);
        c.drawCircle(size/2f, size/2f, size/3f, p);
        
        // Gölgelendirme
        p.setColor(Color.argb(50, 0, 0, 0));
        c.drawCircle(size/2f, size/2f + size/8f, size/3f, p);
        
        // Detaylar (yaprak, sap vb.)
        p.setColor(Color.rgb(50, 100, 50)); // Yeşil sap
        p.setStrokeWidth(10f);
        p.setStyle(Paint.Style.STROKE);
        c.drawLine(size/2f, size/3f, size/2f + 20, size/3f - 40, p);
        
        return bmp;
    }

package com.jai.app.web;
import com.jai.app.data.AppDatabase;
import com.jai.app.data.KnowledgeEntity;
import com.jai.app.tools.ToolResult;
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import java.net.URLEncoder;

public class WebResearchEngine {
    private final AppDatabase db;
    public WebResearchEngine(AppDatabase db) { this.db = db; }
    public ToolResult search(String query) {
        try {
            Document doc = Jsoup.connect("https://html.duckduckgo.com/html/?q=" + URLEncoder.encode(query, "UTF-8")).userAgent("Mozilla/5.0").timeout(10000).get();
            StringBuilder sb = new StringBuilder(); int count = 0;
            for (Element a : doc.select("a.result__a")) {
                String title = a.text(); String href = a.attr("href");
                String snip = "";
                if (a.parent() != null && a.parent().selectFirst(".result__snippet") != null) snip = a.parent().selectFirst(".result__snippet").text();
                sb.append("• ").append(title).append("\n  ").append(snip).append("\n\n");
                db.knowledge().insert(new KnowledgeEntity(query, title + " - " + snip, href));

package com.jai.app.codelab;
public class CodeLabEngine {
    public static class Analysis {
        public final boolean ok; public final String report;
        public Analysis(boolean ok, String report) { this.ok = ok; this.report = report; }
    }
    public Analysis analyze(String code, String lang) {
        if (code == null || code.isEmpty()) return new Analysis(false, "Boş kod");
        if (lang.equals("json")) {
            try { new org.json.JSONObject(code); return new Analysis(true, "Geçerli JSON"); }
            catch (Exception e) { return new Analysis(false, "Geçersiz JSON: " + e.getMessage()); }
        }
        int braces = 0, parens = 0;
        boolean inString = false, inChar = false, inLineComment = false, inBlockComment = false;
        char prev = 0;
        for (char c : code.toCharArray()) {
            if (inLineComment) { if (c == '\n') inLineComment = false; continue; }
            if (inBlockComment) { if (prev == '*' && c == '/') inBlockComment = false; prev = c; continue; }
            if (!inString && !inChar) {
                if (prev == '/' && c == '/') { inLineComment = true; prev = c; continue; }
                if (prev == '/' && c == '*') { inBlockComment = true; prev = c; continue; }
            }
            if (c == '"' && prev != '\\') inString = !inString;
            if (c == '\'' && prev != '\\') inChar = !inChar;
            if (!inString && !inChar) {
                if (c == '{') braces++; else if (c == '}') braces--;
                else if (c == '(') parens++; else if (c == ')') parens--;
            }
            prev = c;
        }
        if (braces == 0 && parens == 0 && !inString) return new Analysis(true, "Denge tamam");
        return new Analysis(false, "Denge bozuk: {}=" + braces + " ()=" + parens + " string=" + inString);
    }
}

// DOSYA 121: app/src/main/java/com/jai/app/trainer/Loss.java
package com.jai.app.trainer;
import com.jai.app.model.Matrix;
public class Loss {
    public static float crossEntropy(Matrix logits, int[] targets) {
        float total = 0;
        for (int i = 0; i < logits.rows; i++) {
            int row = i * logits.cols; float max = Float.NEGATIVE_INFINITY;
            for (int j = 0; j < logits.cols; j++) if (logits.data[row + j] > max) max = logits.data[row + j];
            float sum = 0; for (int j = 0; j < logits.cols; j++) sum += (float) Math.exp(logits.data[row + j] - max);
            total -= (logits.data[row + targets[i]] - max - (float) Math.log(sum));
        }
        return total / logits.rows;
    }
    public static Matrix crossEntropyGrad(Matrix logits, int[] targets) {
        Matrix grad = new Matrix(logits.rows, logits.cols);
        for (int i = 0; i < logits.rows; i++) {
            int row = i * logits.cols; float max = Float.NEGATIVE_INFINITY;
            for (int j = 0; j < logits.cols; j++) if (logits.data[row + j] > max) max = logits.data[row + j];
            float sum = 0; for (int j = 0; j < logits.cols; j++) sum += (float) Math.exp(logits.data[row + j] - max);
            for (int j = 0; j < logits.cols; j++) grad.data[row + j] = (float) Math.exp(logits.data[row + j] - max) / sum;
            grad.data[row + targets[i]] -= 1;
            for (int j = 0; j < logits.cols; j++) grad.data[row + j] /= logits.rows;
        }
        return grad;
    }
}

// DOSYA 123: app/src/main/java/com/jai/app/trainer/Trainer.java
package com.jai.app.trainer;
import com.jai.app.model.Matrix;
import com.jai.app.model.TransformerModel;
public class Trainer {
    public final TransformerModel model;
    public final Optimizer opt;
    public Trainer(TransformerModel m, float lr) { this.model = m; this.opt = new Optimizer(lr); }
    
    public float trainStep(int[] input, int[] targets) {
        Matrix logits = model.forward(input);
        int seq = Math.min(input.length, logits.rows);
        Matrix subLogits = new Matrix(seq, logits.cols);
        for (int i = 0; i < seq; i++) System.arraycopy(logits.data, i * logits.cols, subLogits.data, i * logits.cols, logits.cols);
        int[] subTargets = new int[seq]; System.arraycopy(targets, 0, subTargets, 0, seq);
        
        float loss = Loss.crossEntropy(subLogits, subTargets);
        
        Matrix gradLogits = Loss.crossEntropyGrad(subLogits, subTargets);
        Matrix lastHidden = model.tokenEmb.forward(input); 
        Matrix gradW = Matrix.mul(gradLogits.transpose(), lastHidden);
        
        opt.step(model.lmHead.W, gradW);
        
        return loss;
    }
}

package com.jai.app.trainer;
import com.jai.app.model.Matrix;
import com.jai.app.model.TransformerModel;
import java.io.DataInputStream;
import java.io.InputStream;

public class PtqReader {
    public static TransformerModel read(InputStream in) throws Exception {
        DataInputStream din = new DataInputStream(in);
        int vocab = din.readInt(); int dim = din.readInt(); int layers = din.readInt();
        TransformerModel m = new TransformerModel(vocab, dim, layers, 8, 2304, 1024);
        
        readMatrix(din, m.tokenEmb.W); readMatrix(din, m.posEmb.W);
        for (var b : m.blocks) {
            readMatrix(din, b.attn.qProj.W); readMatrix(din, b.attn.kProj.W);
            readMatrix(din, b.attn.vProj.W); readMatrix(din, b.attn.oProj.W);
            readMatrix(din, b.ffn.fc1.W); readMatrix(din, b.ffn.fc2.W);
        }
        readMatrix(din, m.lmHead.W);
        return m;
    }
    private static void readMatrix(DataInputStream din, Matrix target) throws Exception {
        int rows = din.readInt(); int cols = din.readInt();
        float scale = din.readFloat(); int zp = din.readInt();
        byte[] data = new byte[rows * cols];
        din.readFully(data);
        for (int i = 0; i < data.length; i++) target.data[i] = ((data[i] & 0xFF) - zp) * scale;
    }
}

// DOSYA 132: server/src/main/java/com/jai/server/MessagingServer.java
package com.jai.server;
import org.java_websocket.WebSocket;
import org.java_websocket.handshake.ClientHandshake;
import org.java_websocket.server.WebSocketServer;
import java.net.InetSocketAddress;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
public class MessagingServer extends WebSocketServer {
    private final Map<String, WebSocket> online = new ConcurrentHashMap<>();
    public MessagingServer(int port) { super(new InetSocketAddress(port)); }
    @Override public void onOpen(WebSocket c, ClientHandshake h) {
        String q = h.getResourceDescriptor(); int i = q.indexOf("user=");
        if (i >= 0) online.put(q.substring(i + 5), c);
    }
    @Override public void onClose(WebSocket c, int code, String reason, boolean remote) { online.values().remove(c); }
    @Override public void onMessage(WebSocket c, String msg) {
        try {
            org.json.JSONObject j = new org.json.JSONObject(msg);
            if (j.has("to")) {
                WebSocket t = online.get(j.getString("to"));
                if (t != null && t.isOpen()) t.send(msg);
            }
        } catch (Exception ignored) {}
    }
    @Override public void onError(WebSocket c, Exception ex) {}
    @Override public void onStart() {}
    public static void main(String[] a) { new MessagingServer(8887).start(); }
}
