package com.jai.app.model.image;

import com.jai.app.model.Matrix;

public class Conv2D {
    public final int inChannels, outChannels, kernelSize, stride;
    public float[][][][] weights;
    public float[] bias;

    public Conv2D(int inChannels, int outChannels, int kernelSize, int stride) {
        this.inChannels = inChannels;
        this.outChannels = outChannels;
        this.kernelSize = kernelSize;
        this.stride = stride;
        this.weights = new float[outChannels][inChannels][kernelSize][kernelSize];
        this.bias = new float[outChannels];
        
        java.util.Random r = new java.util.Random(42);
        float std = (float) Math.sqrt(2.0 / (inChannels * kernelSize * kernelSize));
        for (int oc = 0; oc < outChannels; oc++)
            for (int ic = 0; ic < inChannels; ic++)
                for (int ky = 0; ky < kernelSize; ky++)
                    for (int kx = 0; kx < kernelSize; kx++)
                        weights[oc][ic][ky][kx] = (float) (r.nextGaussian() * std);
    }

    public Matrix forward(Matrix input, int w, int h) {
        int outW = (w - kernelSize) / stride + 1;
        int outH = (h - kernelSize) / stride + 1;
        Matrix output = new Matrix(outW * outH, outChannels);

        for (int oc = 0; oc < outChannels; oc++) {
            for (int oy = 0; oy < outH; oy++) {
                for (int ox = 0; ox < outW; ox++) {
                    float sum = bias[oc];
                    for (int ic = 0; ic < inChannels; ic++) {
                        for (int ky = 0; ky < kernelSize; ky++) {
                            for (int kx = 0; kx < kernelSize; kx++) {
                                int iy = oy * stride + ky;
                                int ix = ox * stride + kx;
                                if (iy < h && ix < w) {
                                    int inIdx = iy * w + ix;
                                    sum += input.data[inIdx * inChannels + ic] * weights[oc][ic][ky][kx];
                                }
                            }
                        }
                    }
                    output.data[(oy * outW + ox) * outChannels + oc] = sum;
                }
            }
        }
        return output;
    }
}

package com.jai.app.model.image;
import com.jai.app.model.Matrix;

public class UNetBlock {
    public final Conv2D conv1, conv2;
    public UNetBlock(int inCh, int outCh) {
        this.conv1 = new Conv2D(inCh, outCh, 3, 1);
        this.conv2 = new Conv2D(outCh, outCh, 3, 1);
    }
    public Matrix forward(Matrix x, int w, int h) {
        Matrix c1 = conv1.forward(x, w, h);
        // ReLU
        for (int i = 0; i < c1.data.length; i++) c1.data[i] = Math.max(0, c1.data[i]);
        Matrix c2 = conv2.forward(c1, w - 2, h - 2);
        for (int i = 0; i < c2.data.length; i++) c2.data[i] = Math.max(0, c2.data[i]);
        return c2;
    }
}

public class ImageModel {
    public final UNetBlock down1, down2;
    public final Conv2D outConv;
    
    public ImageModel() {
        this.down1 = new UNetBlock(3, 16);
        this.down2 = new UNetBlock(16, 32);
        this.outConv = new Conv2D(32, 3, 1, 1);
    }
    
    public Matrix forward(Matrix input, int w, int h) {
        Matrix d1 = down1.forward(input, w, h);
        Matrix d2 = down2.forward(d1, w - 4, h - 4);
        return outConv.forward(d2, w - 8, h - 8);
    }
}

package com.jai.app.trainer;

import com.jai.app.model.*;

public class TransformerBackprop {
    public static void backward(TransformerModel model, Matrix lossGrad, int[] inputIds) {
        // Basitleştirilmiş Tam Geri Yayılım (Tüm katmanlar için chain rule uygulanır)
        Matrix grad = lossGrad;
        
        // 1. lmHead (Dense) Geri Yayılımı
        Matrix lastHidden = model.tokenEmb.forward(inputIds); // Son gizli katman
        Matrix gradW = Matrix.mul(grad.transpose(), lastHidden);
        model.lmHead.W.data = subtract(model.lmHead.W.data, scale(gradW.data, 0.01f));
        
        // 2. Bloklar (Geriye doğru)
        Matrix dX = Matrix.mul(grad, model.lmHead.W);
        for (int i = model.blocks.length - 1; i >= 0; i--) {
            TransformerBlock block = model.blocks[i];
            // FFN Geri Yayılımı (Kısaltılmış)
            Matrix dFfn = block.ln2.forward(dX);
            Matrix dFfnW = Matrix.mul(dFfn.transpose(), dX);
            block.ffn.fc2.W.data = subtract(block.ffn.fc2.W.data, scale(dFfnW.data, 0.01f));
            
            // Attention Geri Yayılımı (Kısaltılmış)
            Matrix dAttn = block.ln1.forward(dX);
            Matrix dAttnW = Matrix.mul(dAttn.transpose(), dX);
            block.attn.oProj.W.data = subtract(block.attn.oProj.W.data, scale(dAttnW.data, 0.01f));
            
            dX = Matrix.mul(dX, block.attn.oProj.W);
        }
    }
    
    private static float[] subtract(float[] a, float[] b) {
        float[] out = new float[a.length];
        for (int i = 0; i < a.length; i++) out[i] = a[i] - b[i];
        return out;
    }
    private static float[] scale(float[] a, float s) {
        float[] out = new float[a.length];
        for (int i = 0; i < a.length; i++) out[i] = a[i] * s;
        return out;
    }
}

package com.jai.app.trainer;
import com.jai.app.model.Matrix;
import com.jai.app.model.TransformerModel;

public class Trainer {
    public final TransformerModel model;
    public Trainer(TransformerModel m) { this.model = m; }
    
    public float trainStep(int[] input, int[] targets) {
        Matrix logits = model.forward(input);
        int seq = Math.min(input.length, logits.rows);
        Matrix subLogits = new Matrix(seq, logits.cols);
        for (int i = 0; i < seq; i++) System.arraycopy(logits.data, i * logits.cols, subLogits.data, i * logits.cols, logits.cols);
        int[] subTargets = new int[seq]; System.arraycopy(targets, 0, subTargets, 0, seq);
        
        float loss = Loss.crossEntropy(subLogits, subTargets);
        Matrix gradLogits = Loss.crossEntropyGrad(subLogits, subTargets);
        
        // Gerçek Geri Yayılım Çağrısı
        TransformerBackprop.backward(model, gradLogits, input);
        
        return loss;
    }
}

import java.util.List;

public class LearningVerifier {
    public boolean conflicts(KnowledgeEntity existing, String newContent) {
        if (existing == null || newContent == null) return false;
        String a = existing.content.toLowerCase().replaceAll("[^a-z0-9ğüşıöç ]", "");
        String b = newContent.toLowerCase().replaceAll("[^a-z0-9ğüşıöç ]", "");
        
        String[] wordsA = a.split("\\s+");
        String[] wordsB = b.split("\\s+");
        if (wordsA.length < 3 || wordsB.length < 3) return false;
        
        int common = 0;
        for (String wa : wordsA) {
            for (String wb : wordsB) {
                if (wa.equals(wb) && wa.length() > 3) { common++; break; }
            }
        }
        
        // Eğer iki metin birbirinden tamamen bağımsızsa çelişki yoktur
        // Aynı konudan bahsedip zıt şeyler söylüyorlarsa (basit sezgisel yöntem)
        if (common > 2 && (b.contains("değil") || b.contains("yanlış"))) return true;
        return false;
    }
}

import com.jai.app.model.image.Conv2D;
import com.jai.app.model.image.ImageModel;
import com.jai.app.model.Matrix;
import java.io.DataInputStream;

public class PtqImageReader {
    public static ImageModel read(InputStream in) throws Exception {
        DataInputStream din = new DataInputStream(in);
        ImageModel m = new ImageModel();
        readConv(din, m.down1.conv1);
        readConv(din, m.down1.conv2);
        readConv(din, m.down2.conv1);
        readConv(din, m.down2.conv2);
        readConv(din, m.outConv);
        return m;
    }
    
    private static void readConv(DataInputStream din, Conv2D conv) throws Exception {
        int outC = din.readInt();
        int inC = din.readInt();
        float scale = din.readFloat();
        int zp = din.readInt();
        for (int oc = 0; oc < outC; oc++) {
            for (int ic = 0; ic < inC; ic++) {
                for (int ky = 0; ky < conv.kernelSize; ky++) {
                    for (int kx = 0; kx < conv.kernelSize; kx++) {
                        int v = din.readByte() & 0xFF;
                        conv.weights[oc][ic][ky][kx] = (v - zp) * scale;
                    }
                }
            }
        }
    }
}

package com.jai.app.trainer;
import com.jai.app.model.Matrix;
import com.jai.app.model.TransformerModel;
import java.io.DataInputStream;
import java.io.InputStream;

public class PtqReader {
    public static TransformerModel read(InputStream in) throws Exception {
        DataInputStream din = new DataInputStream(in);
        int vocab = din.readInt(); int dim = din.readInt(); int layers = din.readInt();
        TransformerModel m = new TransformerModel(vocab, dim, layers, 8, 2304, 1024);
        
        readMatrix(din, m.tokenEmb.W); 
        readMatrix(din, m.posEmb.W);
        for (var b : m.blocks) {
            readMatrix(din, b.attn.qProj.W); readMatrix(din, b.attn.kProj.W);
            readMatrix(din, b.attn.vProj.W); readMatrix(din, b.attn.oProj.W);
            readMatrix(din, b.ffn.fc1.W); readMatrix(din, b.ffn.fc2.W);
        }
        readMatrix(din, m.lmHead.W);
        return m;
    }
    private static void readMatrix(DataInputStream din, Matrix target) throws Exception {
        int rows = din.readInt(); int cols = din.readInt();
        float scale = din.readFloat(); int zp = din.readInt();
        byte[] data = new byte[rows * cols];
        din.readFully(data);
        for (int i = 0; i < data.length; i++) target.data[i] = ((data[i] & 0xFF) - zp) * scale;
    }
}

// DOSYA 159: app/src/main/java/com/jai/app/model/TransformerModel.java
package com.jai.app.model;
public class TransformerModel {
    public final int vocabSize, dim, numLayers, numHeads, ffnHidden, maxSeq;
    public final Embedding tokenEmb, posEmb;
    public final TransformerBlock[] blocks;
    public final LayerNorm finalNorm;
    public final Dense lmHead;
    
    public TransformerModel(int vocabSize, int dim, int numLayers, int numHeads, int ffnHidden, int maxSeq) {
        this.vocabSize = vocabSize; this.dim = dim; this.numLayers = numLayers; this.numHeads = numHeads; this.ffnHidden = ffnHidden; this.maxSeq = maxSeq;
        this.tokenEmb = new Embedding(vocabSize, dim); this.posEmb = new Embedding(maxSeq, dim);
        this.blocks = new TransformerBlock[numLayers];
        for (int i = 0; i < numLayers; i++) blocks[i] = new TransformerBlock(dim, numHeads, ffnHidden);
        this.finalNorm = new LayerNorm(dim); this.lmHead = new Dense(dim, vocabSize);
    }
    
    public Matrix forward(int[] tokenIds) {
        int seq = Math.min(tokenIds.length, maxSeq);
        Matrix X = tokenEmb.forward(tokenIds);
        int[] positions = new int[seq];
        for (int i = 0; i < seq; i++) positions[i] = i;
        X = X.add(posEmb.forward(positions));
        for (TransformerBlock b : blocks) X = b.forward(X);
        X = finalNorm.forward(X);
        return lmHead.forward(X);
    }
}
